        import cv2  # noqa
    except Exception:
        missing.append("opencv-python")

        import mss  # noqa

        missing.append("mss")

    if not missing:
        return

    print("[SETUP] Installing missing packages:", ", ".join(missing), flush=True)

        subprocess.check_call(
            [sys.executable, "-m", "pip", "install", "--user"] + missing

        print("Run manually:", flush=True)
        print("  py -m pip install --user numpy opencv-python mss", flush=True)
        input("Press Enter to close...")
        raise SystemExit(1)

    os.execv(sys.executable, [sys.executable] + sys.argv)

def load_reference(key):
    path = REF_DIR / REF_FILES[key]

    if not path.exists():
        print(f"[ERROR] Missing reference: {path}", flush=True)
        raise SystemExit(1)

    img = cv2.imread(str(path), cv2.IMREAD_COLOR)

    if img is None:
        print(f"[ERROR] Could not load reference: {path}", flush=True)
        raise SystemExit(1)

    print(
        f"[REF] {key:16s} {path.name:34s} "
        f"{img.shape[1]}x{img.shape[0]}",
        flush=True,
    )

    return img

def build_fish_shape_mask(image):
    hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)

    # The reference crop has a dark blue background. V>=150 isolates the
    # bright cyan fish silhouette very cleanly.
    mask = cv2.inRange(
        hsv,
        (88, 120, 150),
        (125, 255, 255),
    )

    ys, xs = np.where(mask > 0)

    if len(xs) == 0:
        return np.zeros((32, 32), dtype=np.uint8)

    crop = mask[
        int(ys.min()):int(ys.max()) + 1,
        int(xs.min()):int(xs.max()) + 1,
    ]

    return cv2.resize(
        crop,
        (32, 32),
        interpolation=cv2.INTER_NEAREST,
    )

# use_last_error=True is required for ctypes.get_last_error() to return
# anything. Without it the LastError logged next to every SendInput result is
# always 0, which is exactly the number you need when SendInput fails.
user32 = ctypes.WinDLL("user32", use_last_error=True)

try:
    user32.SetProcessDPIAware()
except Exception:
    pass

ACTIVE_MONITOR = None
TARGET_HWND = None

GA_ROOT = 2


class POINT(ctypes.Structure):
    _fields_ = [
        ("x", wintypes.LONG),
        ("y", wintypes.LONG),
    ]


def hwnd_title(hwnd):
    if not hwnd:
        return ""

    length = user32.GetWindowTextLengthW(hwnd)

    if length <= 0:
        return ""

    buf = ctypes.create_unicode_buffer(length + 1)
    user32.GetWindowTextW(hwnd, buf, length + 1)
    return buf.value


def foreground_title():
    hwnd = user32.GetForegroundWindow()
    return hwnd_title(hwnd) or f"HWND {hwnd}"


def target_window_from_detection(sct, monitor_index, box):
    global TARGET_HWND

    if box is None:
        return None

    mon = sct.monitors[monitor_index]

    x, y, w, h = box

    screen_x = int(mon["left"] + x + w / 2)
    screen_y = int(mon["top"] + y + h / 2)

    pt = POINT(screen_x, screen_y)

    hwnd = user32.WindowFromPoint(pt)

    if hwnd:
        root = user32.GetAncestor(hwnd, GA_ROOT)

        if root:
            hwnd = root

    if hwnd:
        changed = hwnd != TARGET_HWND
        TARGET_HWND = hwnd

        if changed:
            print(
                f"[TARGET] Locked window under detection: "
                f"{hwnd_title(hwnd) or ('HWND ' + str(hwnd))}",
                flush=True,
            )

    return hwnd


def focus_target():
    if not TARGET_HWND:
        return False

    try:
        user32.SetForegroundWindow(TARGET_HWND)
        return True
    except Exception:
        return False


def focus_game():
    # Kept as the shared input hook name.
    focus_target()

INPUT_KEYBOARD = 1
KEYEVENTF_SCANCODE = 0x0008
KEYEVENTF_KEYUP = 0x0002

VK_F8 = 0x77
VK_F9 = 0x78

VK_CODE = {
    "A": 0x41,
    "D": 0x44,
    "E": 0x45,
    "SPACE": 0x20,
}

SCAN_CODE = {
    "A": 0x1E,
    "D": 0x20,
    "E": 0x12,
    "SPACE": 0x39,
}

ULONG_PTR = wintypes.WPARAM


class KEYBDINPUT(ctypes.Structure):
    _fields_ = [
        ("wVk", wintypes.WORD),
        ("wScan", wintypes.WORD),
        ("dwFlags", wintypes.DWORD),
        ("time", wintypes.DWORD),
        ("dwExtraInfo", ULONG_PTR),
    ]


class MOUSEINPUT(ctypes.Structure):
    _fields_ = [
        ("dx", wintypes.LONG),
        ("dy", wintypes.LONG),
        ("mouseData", wintypes.DWORD),
        ("dwFlags", wintypes.DWORD),
        ("time", wintypes.DWORD),
        ("dwExtraInfo", ULONG_PTR),
    ]


class HARDWAREINPUT(ctypes.Structure):
    _fields_ = [
        ("uMsg", wintypes.DWORD),
        ("wParamL", wintypes.WORD),
        ("wParamH", wintypes.WORD),
    ]


class INPUTUNION(ctypes.Union):
    # THE UNION MUST CONTAIN ALL THREE MEMBERS.
    #
    # SendInput's third argument is the size of one INPUT structure, and
    # Windows rejects the call outright unless it matches what the OS expects:
    # 40 bytes on 64-bit Windows, 28 on 32-bit. With only KEYBDINPUT in the
    # union the structure is 32 bytes, so every SendInput call fails with
    # ERROR_INVALID_PARAMETER and returns 0.
    #
    # That is the "SendInput=0/1" in the log. Keys were only ever reaching the
    # game through the legacy keybd_event fallback below it; the scan-code
    # SendInput path - the one that actually matters for games that read raw
    # input - has never once succeeded.
    _fields_ = [
        ("ki", KEYBDINPUT),
        ("mi", MOUSEINPUT),
        ("hi", HARDWAREINPUT),
    ]


class INPUT(ctypes.Structure):
    _anonymous_ = ("u",)
    _fields_ = [
        ("type", wintypes.DWORD),
        ("u", INPUTUNION),
    ]


held = {

def send_input_event(key, down):
    """
    Stronger v7 keyboard injection.

    Sends the physical scan-code through SendInput AND a matching keybd_event.
    This improves compatibility with games that ignore one of the two Windows
    keyboard paths.

    Console reports whether SendInput itself accepted the event.
    """
    focus_target()

    flags = KEYEVENTF_SCANCODE

    if not down:
        flags |= KEYEVENTF_KEYUP

    inp = INPUT(
        type=INPUT_KEYBOARD,
        ki=KEYBDINPUT(
            wVk=0,
            wScan=SCAN_CODE[key],
            dwFlags=flags,
            time=0,
            dwExtraInfo=0,
        ),
    )

    ctypes.set_last_error(0)

    sent = user32.SendInput(
        1,
        ctypes.byref(inp),
        ctypes.sizeof(INPUT),
    )

    send_last_error = ctypes.get_last_error()

    dbg_input(
        key,
        "DOWN" if down else "UP",
        sent,
        send_last_error,
    )

    # Compatibility path.
    kb_flags = 0 if down else KEYEVENTF_KEYUP
    user32.keybd_event(
        VK_CODE[key],
        SCAN_CODE[key],
        kb_flags,
        0,
    )

    print(
        f"[INPUT] {key} {'DOWN' if down else 'UP'} "
        f"SendInput={sent}/1 "
        f"LastError={send_last_error} "
        f"target={TARGET_HWND} "
        f"foreground='{foreground_title()}'",
        flush=True,
    )

    return sent == 1


def key_down(key):
    if key in held and held[key]:
        return

    send_input_event(key, True)

    if key in held:
        held[key] = True

    print(f"[KEY] DOWN {key}", flush=True)


def key_up(key):
    if key in held and not held[key]:
        return

    send_input_event(key, False)

    if key in held:
        held[key] = False

    print(f"[KEY] UP   {key}", flush=True)


def tap(key, duration=0.080):
    print(f"[KEY] TAP  {key}", flush=True)

    send_input_event(key, True)
    time.sleep(duration)
    send_input_event(key, False)


def release_movement():
    key_up("A")
    key_up("D")


def drive(direction):
    if direction == "A":
        if held["D"]:
            key_up("D")

        key_down("A")

    elif direction == "D":
        if held["A"]:
            key_up("A")

        key_down("D")

    else:
        release_movement()


def pressed_once(vk, old_state):
    state = bool(user32.GetAsyncKeyState(vk) & 0x8000)
    return state and not old_state, state

def grab_monitor(sct, monitor_index):
    try:
        shot = sct.grab(sct.monitors[monitor_index])
    except Exception:
        return None

    return np.asarray(shot, dtype=np.uint8)[:, :, :3].copy()


def grab_locked_fishing_region(sct, monitor_index, bar):
    """
    Capture only the fishing UI region instead of the entire monitor.

    Returns:
      (frame, local_bar, crop_origin_xy)

    local_bar uses coordinates inside the returned frame.
    crop_origin_xy is the offset back to monitor coordinates.
    """
    if bar is None:
        return None, None, None

    mon = sct.monitors[monitor_index]
    bx, by, bw, bh = map(int, bar)

    x0 = max(0, bx - FAST_REGION_SIDE_PADDING)
    y0 = max(0, by - FAST_REGION_TOP_PADDING)

    x1 = min(
        int(mon["width"]),
        bx + bw + FAST_REGION_SIDE_PADDING,
    )
    y1 = min(
        int(mon["height"]),
        by + bh + FAST_REGION_BOTTOM_PADDING,
    )

    if x1 <= x0 or y1 <= y0:
        return None, None, None

    region = {
        "left": int(mon["left"] + x0),
        "top": int(mon["top"] + y0),
        "width": int(x1 - x0),
        "height": int(y1 - y0),
    }

    try:
        shot = sct.grab(region)
    except Exception:
        return None, None, None

    frame = np.asarray(
        shot,
        dtype=np.uint8,
    )[:, :, :3].copy()

    local_bar = (
        bx - x0,
        by - y0,
        bw,
        bh,
    )

    return frame, local_bar, (x0, y0)


def monitor_description(sct, monitor_index):

    return (
        f"M{monitor_index} "
        f"{mon['width']}x{mon['height']} "
        f"at ({mon['left']},{mon['top']})"
    )

def multiscale_match(frame, ref_key, roi=None, scales=GENERAL_SCALES):
    H, W = frame.shape[:2]

    if roi is None:
        x0, y0, x1, y1 = 0, 0, W, H
    else:
        x0, y0, x1, y1 = map(int, roi)

        x0 = max(0, min(W, x0))
        x1 = max(0, min(W, x1))
        y0 = max(0, min(H, y0))
        y1 = max(0, min(H, y1))

    if x1 <= x0 or y1 <= y0:
        return 0.0, None, None

    search = prepared_gray(frame[y0:y1, x0:x1])
    templ0 = REF_GRAY[ref_key]

    th0, tw0 = templ0.shape

    best_score = -1.0
    best_box = None
    best_scale = None

    for scale in scales:
        tw = max(8, int(tw0 * scale))
        th = max(8, int(th0 * scale))

        if tw >= search.shape[1] or th >= search.shape[0]:
            continue

        interpolation = (
            cv2.INTER_AREA
            if scale < 1.0
            else cv2.INTER_LINEAR
        )

        templ = cv2.resize(
            templ0,
            (tw, th),
            interpolation=interpolation,
        )

        result = cv2.matchTemplate(
            search,
            templ,
            cv2.TM_CCOEFF_NORMED,
        )

        _, score, _, loc = cv2.minMaxLoc(result)

        if score > best_score:
            best_score = float(score)
            best_box = (
                int(loc[0] + x0),
                int(loc[1] + y0),
                int(tw),
                int(th),
            )
            best_scale = float(scale)

    return best_score, best_box, best_scale

def connected_components(mask, minimum_area=1):
    count, _, stats, centers = cv2.connectedComponentsWithStats(mask)

    result = []

    for index in range(1, count):
        x, y, w, h, area = stats[index]

        if area < minimum_area:
            continue

        cx, cy = centers[index]

        result.append(
            {
                "x": int(x),
                "y": int(y),
                "w": int(w),
                "h": int(h),
                "area": int(area),
                "cx": float(cx),
                "cy": float(cy),
            }
        )

    return result


def find_cyan_bar_candidates(frame):
    H, W = frame.shape[:2]

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Cyan used by the supplied fishing UI.
    cyan = cv2.inRange(
        hsv,
        (88, 105, 65),
        (122, 255, 255),
    )

    # Small close joins anti-aliased cyan border segments.
    kernel = np.ones((3, 3), np.uint8)
    cyan_closed = cv2.morphologyEx(
        cyan,
        cv2.MORPH_CLOSE,
        kernel,
        iterations=1,
    )

    parts = connected_components(cyan_closed, 40)

    candidates = []

    for c in parts:
        x = c["x"]
        y = c["y"]
        w = c["w"]
        h = c["h"]

        aspect = w / max(h, 1)
        density = c["area"] / max(w * h, 1)

        # Fishing UI lives in the lower portion.
        if y < H * 0.34:
            continue

        # Main bar is a substantial fraction of screen width.
        if w < W * 0.18:
            continue

        if w > W * 0.92:
            continue

        if h < max(18, H * 0.020):
            continue

        if h > H * 0.16:
            continue

        if aspect < 5.0:
            continue

        if aspect > 28.0:
            continue

        if density < 0.055:
            continue

        candidates.append(
            {
                "box": (x, y, w, h),
                "area": c["area"],
                "density": density,
                "aspect": aspect,
            }
        )

    # Wider candidates first; final validation uses fish reference + green target.
    candidates.sort(
        key=lambda c: (c["box"][2], c["area"]),
        reverse=True,
    )

    return candidates, cyan


def find_cyan_bar_candidates_dynamic(frame):
    """
    FAST-FOLLOW v3 startup detector.

    No X/Y screen-location assumptions.

    The active monitor is downscaled, cyan UI geometry is segmented, and long
    horizontal candidates are returned in full-resolution coordinates.
    Candidate validation happens immediately with fish tracking, so unrelated
    cyan UI elements are rejected on the same frame.
    """
    H, W = frame.shape[:2]

    scale = STARTUP_BAR_DOWNSCALE

    small_w = max(1, int(W * scale))
    small_h = max(1, int(H * scale))

    small = cv2.resize(
        frame,
        (small_w, small_h),
        interpolation=cv2.INTER_AREA,
    )

    hsv = cv2.cvtColor(
        small,
        cv2.COLOR_BGR2HSV,
    )

    cyan = cv2.inRange(
        hsv,
        (88, 90, 45),
        (125, 255, 255),
    )

    cyan = cv2.morphologyEx(
        cyan,
        cv2.MORPH_CLOSE,
        np.ones((3, 3), np.uint8),
        iterations=1,
    )

    parts = connected_components(
        cyan,
        minimum_area=25,
    )

    candidates = []

    for c in parts:
        x = c["x"]
        y = c["y"]
        w = c["w"]
        h = c["h"]
        area = c["area"]

        aspect = w / max(h, 1)
        density = area / max(w * h, 1)

        # Geometry only. There is deliberately no y-position requirement.
        if w < small_w * 0.14:
            continue

        if w > small_w * 0.97:
            continue

        if h < 8:
            continue

        if h > small_h * 0.20:
            continue

        if aspect < 4.0 or aspect > 32.0:
            continue

        if density < 0.04:
            continue

        full_box = (
            int(round(x / scale)),
            int(round(y / scale)),
            int(round(w / scale)),
            int(round(h / scale)),
        )

        # Prefer longer candidates. Fish+green validation is the final filter.
        rank = (
            -float(w)
            - float(area) * 0.002
            + abs(aspect - 10.0) * 0.3
        )

        candidates.append(
            {
                "rank": rank,
                "box": full_box,
            }
        )

    candidates.sort(
        key=lambda item: item["rank"]
    )

    return candidates[:STARTUP_BAR_MAX_CANDIDATES]


def find_fish_in_bar(frame, bar):

    roi = (

        "box": right_box,
        "scale": right_scale,
        "direction": "right",
        "left_score": left_score,
        "right_score": right_score,
    }


def find_green_zone(frame, bar):
    H, W = frame.shape[:2]
    bx, by, bw, bh = bar

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Bright UI green from the user's exact examples.
    green = cv2.inRange(
        hsv,
        (43, 115, 65),
        (93, 255, 255),
    )

    x0 = max(0, bx)
    x1 = min(W, bx + bw)

    y0 = max(0, int(by - bh * 0.10))
    y1 = min(H, int(by + bh * 1.10))

    crop = green[y0:y1, x0:x1]

    if crop.size == 0:
        return None

    # A vertical target may contain thin border lines plus translucent fill.
    # Closing joins the target into a stable region.
    kernel = np.ones((3, 3), np.uint8)
    crop = cv2.morphologyEx(
        crop,
        cv2.MORPH_CLOSE,
        kernel,
        iterations=1,
    )

    parts = connected_components(crop, 20)

    candidates = []

    for c in parts:
        w = c["w"]
        h = c["h"]
        area = c["area"]

        if h < bh * 0.48:
            continue

        if h > bh * 1.20:
            continue

        if w < 3:
            continue

        if w > bw * 0.22:
            continue

        # Target is vertical / roughly square-ish, never a long progress bar.
        if w / max(h, 1) > 1.55:
            continue

        gx = c["x"] + x0
        gy = c["y"] + y0

        candidates.append(
            {
                "box": (gx, gy, w, h),
                "area": area,
            }
        )

    if not candidates:
        return None

    # Prefer largest coherent green region within the main bar.
    candidates.sort(
        key=lambda c: c["area"],
        reverse=True,
    )

    return candidates[0]["box"]


def detect_progress(frame, bar):
    H, W = frame.shape[:2]
    bx, by, bw, bh = bar

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Lower progress is cyan -> green.
    fill = cv2.inRange(
        hsv,
        (34, 100, 55),
        (124, 255, 255),
    )

    x0 = max(0, int(bx - bw * 0.035))
    x1 = min(W, int(bx + bw * 1.035))

    y0 = max(0, int(by + bh + 2))
    y1 = min(H, int(by + bh + max(35, H * 0.11)))

    crop = fill[y0:y1, x0:x1]

    if crop.size == 0:
        return 0.0, None

    row_counts = (crop > 0).sum(axis=1)

    if not row_counts.size:
        return 0.0, None

    best_row = int(np.argmax(row_counts))

    if row_counts[best_row] < max(20, bw * 0.06):
        return 0.0, None

    slab_top = max(0, best_row - 5)
    slab_bottom = min(crop.shape[0], best_row + 6)
    slab = crop[slab_top:slab_bottom]

    ys, xs = np.nonzero(slab)

    if len(xs) < 20:
        return 0.0, None

    left = float(np.percentile(xs, 1)) + x0
    right = float(np.percentile(xs, 99)) + x0

    ratio = (right - bx) / max(float(bw), 1.0)
    ratio = max(0.0, min(1.05, ratio))

    progress_box = (
        int(left),
        int(y0 + slab_top),
        int(max(1.0, right - left)),
        int(max(1, slab_bottom - slab_top)),
    )

    return ratio, progress_box


def remap_box(box, origin):
    if box is None:
        return None

    ox, oy = origin
    x, y, w, h = box

    return (
        int(x + ox),
        int(y + oy),
        int(w),
        int(h),
    )


def track_green_target_borders(
    frame,
    local_bar,
    previous_zone=None,
):
    """
    FAST-FOLLOW v3 green tracker.

    Primary signal = the two bright, almost full-height neon-green vertical
    borders. Their separation is the target's current width, so the target may
    become wider/narrower without any fixed reference-width assumption.

    A broad-green fallback is used only if a border is temporarily covered.
    """
    H, W = frame.shape[:2]
    bx, by, bw, bh = map(int, local_bar)

    if bw <= 20 or bh <= 10:
        return None

    hsv = cv2.cvtColor(
        frame,
        cv2.COLOR_BGR2HSV,
    )

    x0 = max(0, bx + 2)
    x1 = min(W, bx + bw - 2)
    y0 = max(0, by + 2)
    y1 = min(H, by + bh - 2)

    if x1 <= x0 or y1 <= y0:
        return None

    inner_h = y1 - y0

    bright = cv2.inRange(
        hsv,
        GREEN_BORDER_HSV_LOW,
        GREEN_BORDER_HSV_HIGH,
    )[y0:y1, x0:x1]

    column_counts = (
        (bright > 0)
        .sum(axis=0)
        .astype(np.float32)
    )

    required = max(
        3.0,
        inner_h * GREEN_BORDER_MIN_COLUMN_FRACTION,
    )

    active = np.where(
        column_counts >= required
    )[0]

    groups = []

    if len(active):
        s = p = int(active[0])

        for value in active[1:]:
            value = int(value)

            if value <= p + 2:
                p = value
            else:
                groups.append((s, p))
                s = p = value

        groups.append((s, p))

    borders = []

    for g0, g1 in groups:
        segment = column_counts[g0:g1 + 1]

        if not len(segment):
            continue

        peak = g0 + int(np.argmax(segment))

        borders.append(
            {
                "x": float(x0 + peak),
                "strength": float(segment.max()),
            }
        )

    previous_center = None
    previous_width = None

    if previous_zone is not None:
        px, py, pw, ph = previous_zone
        previous_center = px + pw / 2.0
        previous_width = float(pw)

    pairs = []

    for i in range(len(borders)):
        for j in range(i + 1, len(borders)):
            left = borders[i]
            right = borders[j]

            width = right["x"] - left["x"]

            if width < 4.0:
                continue

            if width > bw * GREEN_TARGET_MAX_WIDTH_RATIO:
                continue

            center = (
                left["x"]
                + right["x"]
            ) / 2.0

            # Continuity is only a preference, never a hard jump rejection.
            continuity = 0.0

            if previous_center is not None:
                continuity += (
                    abs(center - previous_center) * 0.22
                )

                if previous_width is not None:
                    continuity += (
                        abs(width - previous_width) * 0.015
                    )

            strength_bonus = (
                left["strength"]
                + right["strength"]
            ) * 0.08

            score = (
                continuity
                - strength_bonus
            )

            pairs.append(
                (
                    score,
                    left["x"],
                    right["x"],
                    center,
                    width,
                )
            )

    if pairs:
        pairs.sort(
            key=lambda item: item[0]
        )

        _, left_x, right_x, center, width = pairs[0]

        return {
            "box": (
                int(round(left_x)),
                int(y0),
                max(
                    1,
                    int(round(
                        right_x - left_x
                    )),
                ),
                int(inner_h),
            ),
            "center_x": float(center),
            "width": float(width),
            "mode": "bright-border-pair",
        }

    # -------------------------------------------------------------
    # Fallback: broad green full-height span
    # -------------------------------------------------------------
    broad = cv2.inRange(
        hsv,
        (45, 100, 65),
        (90, 255, 255),
    )[y0:y1, x0:x1]

    counts = (
        (broad > 0)
        .sum(axis=0)
        .astype(np.float32)
    )

    active = np.where(
        counts >= max(
            3.0,
            inner_h * 0.45,
        )
    )[0]

    if len(active):
        runs = []
        s = p = int(active[0])

        for value in active[1:]:
            value = int(value)

            if value <= p + 3:
                p = value
            else:
                runs.append((s, p))
                s = p = value

        runs.append((s, p))

        spans = []

        for i in range(len(runs)):
            for j in range(i, len(runs)):
                left_local = runs[i][0]
                right_local = runs[j][1]

                width = (
                    right_local
                    - left_local
                    + 1
                )

                if width < 4:
                    continue

                if width > bw * GREEN_TARGET_MAX_WIDTH_RATIO:
                    break

                left_x = x0 + left_local
                right_x = x0 + right_local
                center = (
                    left_x + right_x
                ) / 2.0

                covered = sum(
                    runs[k][1]
                    - runs[k][0]
                    + 1
                    for k in range(i, j + 1)
                )

                coverage = (
                    covered
                    / max(
                        float(width),
                        1.0,
                    )
                )

                if coverage < 0.32:
                    continue

                if previous_center is not None:
                    score = (
                        abs(center - previous_center)
                        + (1.0 - coverage) * 6.0
                    )
                else:
                    # Fresh round target starts centered.
                    score = (
                        abs(
                            center
                            - (bx + bw / 2.0)
                        )
                        + (1.0 - coverage) * 6.0
                    )

                spans.append(
                    (
                        score,
                        left_x,
                        right_x,
                        center,
                        float(width),
                    )
                )

        if spans:
            spans.sort(
                key=lambda item: item[0]
            )

            _, left_x, right_x, center, width = spans[0]

            return {
                "box": (
                    int(round(left_x)),
                    int(y0),
                    max(
                        1,
                        int(round(
                            right_x - left_x
                        )),
                    ),
                    int(inner_h),
                ),
                "center_x": float(center),
                "width": float(width),
                "mode": "broad-green-fallback",
            }

    if previous_zone is not None:
        px, py, pw, ph = previous_zone

        return {
            "box": previous_zone,
            "center_x": float(
                px + pw / 2.0
            ),
            "width": float(pw),
            "mode": "previous-green-grace",
        }

    return None



def fast_match_fish_in_locked_bar(
    frame,
    local_bar,
    previous_scale=None,
):
    """
    Exact fish-left/fish-right template matching every active tracker frame,
    but only inside the tiny known bar and at a narrow scale range.
    """
    H, W = frame.shape[:2]
    bx, by, bw, bh = map(int, local_bar)

    roi = (
        max(0, bx),
        max(0, int(by - bh * 0.10)),
        min(W, bx + bw),
        min(H, int(by + bh * 1.10)),
    )

    if previous_scale is None:
        # Calibrated default from the user's examples.
        base_scale = 0.54
    else:
        base_scale = float(previous_scale)

    scales = tuple(
        max(
            0.28,
            min(
                1.40,
                base_scale * factor,
            ),
        )
        for factor in FISH_ACTIVE_SCALE_FACTORS
    )

    left_score, left_box, left_scale = multiscale_match(
        frame,
        "fish_left",
        roi,
        scales,
    )

    right_score, right_box, right_scale = multiscale_match(
        frame,
        "fish_right",
        roi,
        scales,
    )

    if left_score >= right_score:
        return {
            "score": float(left_score),
            "box": left_box,
            "scale": left_scale,
            "direction": "left",
            "left_score": float(left_score),
            "right_score": float(right_score),
        }

    return {
        "score": float(right_score),
        "box": right_box,
        "scale": right_scale,
        "direction": "right",
        "left_score": float(left_score),
        "right_score": float(right_score),
    }


def fast_track_minigame(
    frame,
    local_bar,
    previous_ui=None,
    allow_startup_center=False,
):
    """
    FAST-FOLLOW v3.

    Fish detection:
      - high-value cyan pixels only
      - candidate silhouette compared against supplied fish-left/right shapes
      - no hard "fish cannot jump this far" rejection

    This prevents cyan scenery/background inside the transparent bar from being
    chosen as the fish and also allows the real fish to make a sudden move.

    Green:
      - live bright-border pair
      - fully dynamic width
    """
    if frame is None or local_bar is None:
        return {
            "valid": False,
            "control_ready": False,
            "bar": local_bar,
        }

    H, W = frame.shape[:2]
    bx, by, bw, bh = map(int, local_bar)

    if bw <= 20 or bh <= 10:
        return {
            "valid": False,
            "control_ready": False,
            "bar": local_bar,
        }

    previous_fish_x = None
    previous_zone = None
    previous_scale = None

    if previous_ui:
        previous_fish_x = previous_ui.get(
            "fish_center_x"
        )
        previous_zone = previous_ui.get(
            "zone"
        )
        previous_scale = previous_ui.get(
            "fish_scale"
        )

    x0 = max(0, bx + 3)
    x1 = min(W, bx + bw - 3)
    y0 = max(0, by + 3)
    y1 = min(H, by + bh - 3)

    # Convert ONLY the bar interior, not the whole frame.
    #
    # In ARMED_FAST this function is called once per bar candidate against the
    # full-screen screenshot, so converting the entire 1920x1080 frame to HSV
    # was being paid over again for every candidate - several milliseconds each
    # time, for a region that is a few percent of the image.
    if x1 > x0 and y1 > y0:
        inner_hsv = cv2.cvtColor(
            frame[y0:y1, x0:x1],
            cv2.COLOR_BGR2HSV,
        )

        inner_cyan = cv2.inRange(
            inner_hsv,
            FISH_BRIGHT_HSV_LOW,
            FISH_BRIGHT_HSV_HIGH,
        )
    else:
        inner_hsv = None
        inner_cyan = None

    fish_box = None
    fish_center_x = None
    fish_score = 0.0
    fish_direction = None
    fish_scale = None
    fish_source = None
    left_score = 0.0
    right_score = 0.0

    if inner_cyan is not None:
        inner = inner_cyan

        parts = connected_components(
            inner,
            minimum_area=20,
        )

        candidates = []

        expected_h = max(
            8.0,
            bh * 0.58,
        )

        expected_w = max(
            8.0,
            bh * 0.66,
        )

        # Resolution-relative size window (see FAST_FISH_*_RATIO above).
        min_fish_area = max(
            FAST_FISH_ABS_MIN_AREA,
            FAST_FISH_MIN_AREA_RATIO * bh * bh,
        )

        max_fish_area = FAST_FISH_MAX_AREA_RATIO * bh * bh

        min_fish_side = max(6.0, bh * 0.16)

        for c in parts:
            w = c["w"]
            h = c["h"]
            area = c["area"]

            if area < min_fish_area or area > max_fish_area:
                continue

            if w < min_fish_side or h < min_fish_side:
                continue

            if h > bh * 0.92:
                continue

            if w > bh * 1.65:
                continue

            aspect = (
                w / max(h, 1)
            )

            if aspect < 0.60 or aspect > 1.85:
                continue

            # Fish is vertically around the middle of the bar.
            cx = float(
                x0 + c["cx"]
            )

            cy = float(
                y0 + c["cy"]
            )

            if abs(
                cy
                - (by + bh / 2.0)
            ) > bh * 0.40:
                continue

            candidate_mask = inner[
                c["y"]:c["y"] + h,
                c["x"]:c["x"] + w,
            ]

            norm = cv2.resize(
                candidate_mask,
                (32, 32),
                interpolation=cv2.INTER_NEAREST,
            )

            norm_bool = norm > 0
            left_bool = FISH_SHAPE_LEFT > 0
            right_bool = FISH_SHAPE_RIGHT > 0

            left_union = np.logical_or(
                norm_bool,
                left_bool,
            ).sum()

            right_union = np.logical_or(
                norm_bool,
                right_bool,
            ).sum()

            left_iou = (
                np.logical_and(
                    norm_bool,
                    left_bool,
                ).sum()
                / max(
                    float(left_union),
                    1.0,
                )
            )

            right_iou = (
                np.logical_and(
                    norm_bool,
                    right_bool,
                ).sum()
                / max(
                    float(right_union),
                    1.0,
                )
            )

            shape_score = max(
                left_iou,
                right_iou,
            )

            if shape_score < FISH_SHAPE_MIN_SCORE:
                continue

            direction = (
                "left"
                if left_iou >= right_iou
                else "right"
            )

            size_error = (
                abs(h - expected_h)
                / expected_h
                + abs(w - expected_w)
                / expected_w
            )

            continuity = 0.0

            if previous_fish_x is not None:
                # Weak preference only. Sudden real fish jumps are allowed.
                continuity = (
                    abs(cx - previous_fish_x)
                    / max(
                        float(bw),
                        1.0,
                    )
                    * 0.25
                )

            rank = (
                -shape_score * 4.0
                + size_error * 0.45
                + continuity
            )

            candidates.append(
                (
                    rank,
                    shape_score,
                    (
                        int(x0 + c["x"]),
                        int(y0 + c["y"]),
                        int(w),
                        int(h),
                    ),
                    cx,
                    direction,
                )
            )

        if candidates:
            candidates.sort(
                key=lambda item: item[0]
            )

            (
                _,
                fish_score,
                fish_box,
                fish_center_x,
                fish_direction,
            ) = candidates[0]

            fish_source = "shape-cyan"

    # Same-frame exact template fallback if the shape component is unavailable.
    if fish_box is None:
        fish_ref = fast_match_fish_in_locked_bar(
            frame,
            local_bar,
            previous_scale=previous_scale,
        )

        left_score = float(
            fish_ref["left_score"]
        )

        right_score = float(
            fish_ref["right_score"]
        )

        if (
            fish_ref["box"] is not None
            and fish_ref["score"]
            >= FISH_ACTIVE_MIN_SCORE
        ):
            fish_box = fish_ref["box"]
            fish_score = float(
                fish_ref["score"]
            )
            fish_direction = fish_ref["direction"]
            fish_scale = fish_ref["scale"]
            fish_center_x = (
                fish_box[0]
                + fish_box[2] / 2.0
            )
            fish_source = "template"

    if fish_box is None or fish_center_x is None:
        return {
            "valid": False,
            "control_ready": False,
            "bar": local_bar,
            "fish": None,
            "zone": None,
            "fish_score": 0.0,
            "fish_left_score": left_score,
            "fish_right_score": right_score,
            "fish_source": None,
        }

    green = track_green_target_borders(
        frame,
        local_bar,
        previous_zone=previous_zone,
    )

    if (
        green is None
        and allow_startup_center
    ):
        # The user confirmed a fresh round starts with green centered.
        center = (
            bx + bw / 2.0
        )

        inferred_width = max(
            8,
            int(bh * 0.75),
        )

        green = {
            "box": (
                int(
                    center
                    - inferred_width / 2.0
                ),
                int(by + 2),
                int(inferred_width),
                max(
                    1,
                    int(bh - 4),
                ),
            ),
            "center_x": float(center),
            "width": float(inferred_width),
            "mode": "startup-center",
        }

    if green is None:
        return {
            "valid": False,
            "control_ready": False,
            "bar": local_bar,
            "fish": fish_box,
            "zone": None,
            "fish_score": fish_score,
            "fish_left_score": left_score,
            "fish_right_score": right_score,
            "fish_direction": fish_direction,
            "fish_scale": fish_scale,
            "fish_source": fish_source,
            "fish_center_x": float(
                fish_center_x
            ),
        }

    zone_box = green["box"]
    zone_center_x = float(
        green["center_x"]
    )

    zx, zy, zw, zh = zone_box

    return {
        "valid": True,
        "control_ready": True,
        "bar": local_bar,
        "fish": fish_box,
        "fish_score": float(
            fish_score
        ),
        "fish_left_score": left_score,
        "fish_right_score": right_score,
        "fish_direction": fish_direction,
        "fish_scale": fish_scale,
        "fish_source": fish_source,
        "zone": zone_box,
        "green_mode": green["mode"],
        "fish_center_x": float(
            fish_center_x
        ),
        "zone_center_x": float(
            zone_center_x
        ),
        "error": float(
            fish_center_x
            - zone_center_x
        ),
        "inside": bool(
            fish_center_x
            >= zx + zw * 0.03
            and fish_center_x
            <= zx + zw * 0.97
        ),
        "progress": 0.0,
        "progress_box": None,
        "ready_score": 0.0,
        "ready_box": None,
        "ready_scale": None,
        "ready_ref": None,
    }



def detect_minigame(frame):
    candidates, cyan_mask = find_cyan_bar_candidates(frame)

    best_partial = None

    for candidate in candidates:
        bar = candidate["box"]

        fish = find_fish_in_bar(frame, bar)
        zone = find_green_zone(frame, bar)

        partial = {
            "bar": bar,
            "bar_aspect": candidate["aspect"],
            "bar_density": candidate["density"],
            "fish": fish["box"],
            "fish_score": fish["score"],
            "fish_left_score": fish["left_score"],
            "fish_right_score": fish["right_score"],
            "fish_direction": fish["direction"],
            "fish_scale": fish["scale"],
            "zone": zone,
            "valid": False,
        }

        if (
            best_partial is None
            or partial["fish_score"] > best_partial["fish_score"]
        ):
            best_partial = partial

        # A valid minigame MUST satisfy the supplied fish reference
        # and green target inside the same cyan bar.
        if fish["score"] < FISH_THRESHOLD:
            continue

        if fish["box"] is None:
            continue

        if zone is None:
            continue

        fx, fy, fw, fh = fish["box"]
        zx, zy, zw, zh = zone

        fish_center_x = fx + fw / 2.0
        zone_center_x = zx + zw / 2.0

        error = fish_center_x - zone_center_x

        # Give a small tolerance at the edges. The fish should be fully
        # controllable without requiring pixel-perfect containment.
        inside = (
            fish_center_x >= zx + zw * 0.04
            and fish_center_x <= zx + zw * 0.96
        )

        progress, progress_box = detect_progress(frame, bar)

        ready_score, ready_box, ready_scale, ready_ref = detect_ready(
            frame,
            bar,
        )

        return {
            "valid": True,
            "bar": bar,
            "fish": fish["box"],
            "fish_score": fish["score"],
            "fish_left_score": fish["left_score"],
            "fish_right_score": fish["right_score"],
            "fish_direction": fish["direction"],
            "fish_scale": fish["scale"],
            "zone": zone,
            "fish_center_x": float(fish_center_x),
            "zone_center_x": float(zone_center_x),
            "error": float(error),
            "inside": bool(inside),
            "progress": float(progress),
            "progress_box": progress_box,
            "ready_score": float(ready_score),
            "ready_box": ready_box,
            "ready_scale": ready_scale,
            "ready_ref": ready_ref,
        }

    if best_partial is not None:
        return best_partial

    return {
        "valid": False,
        "bar": None,
        "fish": None,
        "fish_score": 0.0,
        "fish_left_score": 0.0,
        "fish_right_score": 0.0,
        "fish_direction": None,
        "fish_scale": None,
        "zone": None,
    }

_IMAGE_WRITE_QUEUE = queue.Queue(maxsize=128)


def _image_writer_worker():
    while True:
        item = _IMAGE_WRITE_QUEUE.get()

        if item is None:
            _IMAGE_WRITE_QUEUE.task_done()
            return

        path, image = item

        try:
            cv2.imwrite(
                str(path),
                image,
            )
        except Exception:
            pass

        _IMAGE_WRITE_QUEUE.task_done()


_IMAGE_WRITER_THREAD = threading.Thread(
    target=_image_writer_worker,
    name="autofish-image-writer",
    daemon=True,
)

_IMAGE_WRITER_THREAD.start()


def enqueue_image_write(path, image):
    try:
        _IMAGE_WRITE_QUEUE.put_nowait(
            (
                Path(path),
                image.copy(),
            )
        )
        return True
    except queue.Full:
        return False

DETECTION_DIR = BASE / "detections"
DETECTION_DIR.mkdir(exist_ok=True)

_detection_sequence = 0
_last_detection_save = {}


def trim_detection_folder():
    files = sorted(
        DETECTION_DIR.glob("*.png"),
        key=lambda p: p.stat().st_mtime,
    )

    while len(files) > DETECTION_MAX_FILES:
        victim = files.pop(0)

        try:
            victim.unlink()
        except Exception:
            break


def save_reference_hit(
    frame,
    monitor_index,
    ref_name,
    score,
    box,
    threshold,
):
    """
    Save the EXACT matched rectangle, not the whole screenshot.

    A short cooldown prevents one static reference from filling all 100 files
    instantly while still producing a chronological detection trail.
    """
    global _detection_sequence

    if box is None or score < threshold:
        return

    now = time.monotonic()
    last = _last_detection_save.get((monitor_index, ref_name), 0.0)

    if now - last < DETECTION_SAVE_COOLDOWN:
        return

    _last_detection_save[(monitor_index, ref_name)] = now

    H, W = frame.shape[:2]
    x, y, w, h = map(int, box)

    x0 = max(0, x)
    y0 = max(0, y)
    x1 = min(W, x + w)
    y1 = min(H, y + h)

    if x1 <= x0 or y1 <= y0:
        return

    crop = frame[y0:y1, x0:x1].copy()

    _detection_sequence += 1

    safe_name = ref_name.replace("/", "_").replace("\\", "_")

    filename = (
        f"{_detection_sequence:04d}_M{monitor_index}_"
        f"{safe_name}_score-{score:.3f}.png"
    )

    path = DETECTION_DIR / filename

    queued = enqueue_image_write(
        path,
        crop,
    )

    if queued:
        # Folder trimming is deliberately not performed in the control thread.
        # The max-100 cleanup is cheap enough to do only on non-critical saves.
        print(
            f"[SNAPSHOT] Queued exact hit crop: {filename} "
            f"box=({x},{y},{w},{h})",
            flush=True,
        )

def dbg_ui_reason(ui):
    if ui is None:
        return "ui_none"

    if not ui.get("valid", False):
        if ui.get("fish") is None:
            return "fish_missing"

        if ui.get("zone") is None:
            return "green_missing"

        return "invalid_other"

    if not ui.get("control_ready", True):
        return "control_not_ready"

    return "ok"

            "kind": kind,
            **data,
        }
    )

def dbg_ring_image(
    image,
    kind,
    meta,
    scale=1.0,
):
    global _DBG_FRAME_SEQ

    if not FORENSIC_DEBUG or image is None:
        return

    if scale != 1.0:
        image = cv2.resize(
            image,
            None,
            fx=scale,
            fy=scale,
            interpolation=cv2.INTER_AREA,
        )

    # Compress in RAM. No disk I/O.
    ok, encoded = cv2.imencode(
        ".jpg",
        image,
        [int(cv2.IMWRITE_JPEG_QUALITY), 88],
    )

    if not ok:
        return

    _DBG_FRAME_SEQ += 1

    _DBG_RING.append(
        {
            "seq": _DBG_FRAME_SEQ,
            "kind": kind,
            "jpg": encoded.tobytes(),
            "meta": dict(meta),
        }
    )


def dbg_maybe_startup_image(image, meta):
    global _DBG_LAST_STARTUP_IMAGE

    if not FORENSIC_DEBUG:
        return

    now_dbg = time.monotonic()

    if (
        now_dbg - _DBG_LAST_STARTUP_IMAGE
        < DBG_STARTUP_IMAGE_INTERVAL

    _DBG_LAST_STARTUP_IMAGE = now_dbg

    dbg_ring_image(
        image,
        "startup",
        meta,
        scale=DBG_STARTUP_IMAGE_SCALE,
    )

def dbg_freeze(reason, **data):
    """
    Preserve the recent visual ring IN RAM. No disk write happens here.
    """
    if not FORENSIC_DEBUG:
        return

    dbg_event(
        "freeze",
        reason=reason,
        **data,
    )

    if len(_DBG_FROZEN) >= DBG_MAX_FROZEN_FAILURES:
        return

    _DBG_FROZEN.append(
        {
            "reason": reason,
            "data": dict(data),
            "frames": list(_DBG_RING),
        }
    )


def _write_csv(path, rows):
    rows = list(rows)

    if not rows:
        return

    fields = []

    for row in rows:
        for key in row.keys():
            if key not in fields:
                fields.append(key)

    with open(
        path,
        "w",
        newline="",
        encoding="utf-8",
    ) as f:
        writer = csv.DictWriter(
            f,
            fieldnames=fields,
        )

        writer.writeheader()
        writer.writerows(rows)


def dbg_finalize():
    """
    Write all RAM diagnostics only when the program exits, then create one
    ready-to-send ZIP.
    """
    if not FORENSIC_DEBUG:
        return None

    DBG_SESSION.mkdir(
        parents=True,
        exist_ok=True,
    )

    try:
        (DBG_SESSION / "events.json").write_text(
            json.dumps(
                list(_DBG_EVENTS),
                ensure_ascii=False,
                indent=2,
                default=str,
            ),
            encoding="utf-8",
        )

        _write_csv(
            DBG_SESSION / "tracker.csv",
            _DBG_TRACK,
        )

        _write_csv(
            DBG_SESSION / "startup_candidates.csv",
            _DBG_STARTUP,
        )

        _write_csv(
            DBG_SESSION / "input.csv",
            _DBG_INPUT,
        )

        summary = (
            "AUTOFISHING FORENSIC DEBUG\n"
            "===========================\n\n"
            "Send the generated debug_session_*.zip back to ChatGPT.\n\n"
            "tracker.csv:\n"
            "  every active A/D decision and fish/green measurement\n\n"
            "startup_candidates.csv:\n"
            "  every cyan-bar candidate checked before tracking started\n\n"
            "input.csv:\n"

        )

        (DBG_SESSION / "README.txt").write_text(
            summary,
            encoding="utf-8",
        )

        # Frozen failure evidence.
        for index, frozen in enumerate(
            _DBG_FROZEN,
            start=1,
        ):
            safe_reason = "".join(
                ch if ch.isalnum() or ch in "-_" else "_"
                for ch in frozen["reason"]
            )

            failure_dir = (
                DBG_SESSION
                / f"failure_{index:02d}_{safe_reason}"
            )

            failure_dir.mkdir(
                parents=True,
                exist_ok=True,
            )

            manifest = []

            for frame in frozen["frames"]:
                filename = (
                    f"{frame['seq']:06d}_"
                    f"{frame['kind']}.jpg"
                )

                (failure_dir / filename).write_bytes(
                    frame["jpg"]
                )

                manifest.append(
                    {
                        "filename": filename,
                        **frame["meta"],
                    }
                )

            (failure_dir / "manifest.json").write_text(
                json.dumps(
                    {
                        "reason": frozen["reason"],
                        "failure_data": frozen["data"],
                        "frames": manifest,
                    },
                    ensure_ascii=False,
                    indent=2,
                    default=str,
                ),
                encoding="utf-8",
            )

        # Also preserve the last live ring even if no explicit failure froze it.
        tail_dir = DBG_SESSION / "last_live_ring"
        tail_dir.mkdir(
            parents=True,
            exist_ok=True,
        )

        tail_manifest = []

        for frame in list(_DBG_RING):
            filename = (
                f"{frame['seq']:06d}_"
                f"{frame['kind']}.jpg"
            )

            (tail_dir / filename).write_bytes(
                frame["jpg"]
            )

            tail_manifest.append(
                {
                    "filename": filename,
                    **frame["meta"],
                }
            )

        (tail_dir / "manifest.json").write_text(
            json.dumps(
                tail_manifest,
                ensure_ascii=False,
                indent=2,
                default=str,
            ),
            encoding="utf-8",
        )

        zip_path = (
            BASE
            / f"debug_session_{_DBG_STAMP}.zip"
        )

        if zip_path.exists():
            zip_path.unlink()

        with zipfile.ZipFile(
            zip_path,
            "w",
            compression=zipfile.ZIP_DEFLATED,
        ) as z:
            for p in DBG_SESSION.rglob("*"):
                if p.is_file():
                    z.write(
                        p,
                        p.relative_to(DBG_SESSION),
                    )

        print(
            f"[DEBUG-PACK] Created: {zip_path}",
            flush=True,
        )

        return zip_path

    except Exception as exc:
        print(
            f"[DEBUG-PACK] ERROR: {exc}",
            flush=True,
        )

        return None

                    # The sign of CURRENT error is the only thing allowed to

                    #
                    #   fish left of green  -> A
                    #   fish right of green -> D
                    #
                    # Relative-error velocity is used only for EARLY RELEASE
                    # while the target is rapidly approaching the center. It
                    # can never command the opposite direction, which removes
                    # the old "randomly shoots away" failure mode.
                    #
                    error = (

                        - green_x
                    )

                    zone_width = float(

                        CENTER_RELEASE_MIN_PX,

                        previous_control_error is not None
                        and previous_control_time is not None

                            current_time
                            - previous_control_time
                        )

                        if control_dt > 0.0001:
                            instant_rel_v = (
                                error
                                - previous_control_error
                            ) / control_dt

                                -RELATIVE_ERROR_MAX_PX_S,
                                min(
                                    RELATIVE_ERROR_MAX_PX_S,
                                    instant_rel_v,
                                ),
                            )

                            relative_error_velocity = (
                                RELATIVE_ERROR_ALPHA
                                * instant_rel_v
                                + (
                                    1.0
                                    - RELATIVE_ERROR_ALPHA
                                )
                                * relative_error_velocity
                            )

                    previous_control_error = error
                    previous_control_time = current_time

                    predicted_for_brake = (
                        error
                        + relative_error_velocity
                        * BRAKE_LOOKAHEAD_SECONDS
                    )

                    desired = "NONE"

                    if error < -enter_band:
                        # Correct direction is unconditionally LEFT/A.
                        # If we're already closing rapidly, release a little
                        # early instead of overshooting and oscillating.
                        if (
                            relative_error_velocity > 0
                            and predicted_for_brake
                            >= -release_band
                        ):
                            desired = "NONE"
                            drive(None)
                        else:
                            desired = "A"
                            drive("A")

                    elif error > enter_band:
                        # Correct direction is unconditionally RIGHT/D.
                        if (
                            relative_error_velocity < 0
                            and predicted_for_brake
                            <= release_band
                        ):
                            desired = "NONE"
                            drive(None)
                        else:
                            desired = "D"
                            drive("D")

                    else:
                        desired = "NONE"
                        drive(None)

                    previous_fish_x = fish_x
                    previous_fish_time = current_time
                    fish_velocity = 0.0
                    predicted_fish_x = fish_x
                    lead = 0.0

                    dbg_track_row(
                        round_number,
                        ACTIVE_MONITOR,
                        state,
                        ui,
                        desired,
                        fast_track_failures,
                        capture_ms=dbg_capture_ms,
                        detect_ms=dbg_detect_ms,
                        loop_ms=(
                            time.perf_counter()
                            - loop_started
                        ) * 1000.0,
                    )

                    dbg_maybe_active_image(
                        fast_frame,
                        {
                            "round": round_number,
                            "state": state,
                            "monitor": ACTIVE_MONITOR,
                            "reason": dbg_ui_reason(ui),
                            "fish_source": ui.get("fish_source"),
                            "fish_score": ui.get("fish_score"),
                            "fish_x": fish_x,
                            "green_mode": ui.get("green_mode"),
                            "green_x": green_x,
                            "green_w": zone_width,
                            "error": error,
                            "desired_key": desired,
                            "held_A": held["A"],
                            "held_D": held["D"],
                            "failures": fast_track_failures,
                            "capture_ms": dbg_capture_ms,
                            "detect_ms": dbg_detect_ms,
                        },
                    )

                    raw_inside = bool(
                        fish_x
                        >= ui["zone"][0]
                        + ui["zone"][2] * 0.04
                        and fish_x
                        <= ui["zone"][0]
                        + ui["zone"][2] * 0.96
                    )

                    # -----------------------------------------------------
                    # Progress is cheap; sample often.
                    # Ready template is heavier; sample less often.
                    # -----------------------------------------------------
                    if (
                        now - last_progress_check
                        >= PROGRESS_CHECK_INTERVAL
                    ):
                        (
                            cached_progress,
                            local_progress_box,
                        ) = detect_progress(
                            fast_frame,
                            local_bar,
                        )

                        cached_progress_box = remap_box(
                            local_progress_box,
                            crop_origin,
                        )

                        last_progress_check = now

                    if (
                        now - last_ready_check
                        >= READY_CHECK_INTERVAL
                    ):
                        (
                            cached_ready_score,
                            local_ready_box,
                            _,
                            cached_ready_ref,
                        ) = detect_ready(
                            fast_frame,
                            local_bar,
                        )

                        cached_ready_box = remap_box(
                            local_ready_box,
                            crop_origin,
                        )

                        last_ready_check = now

                    # Exact fish reference is already matched every
                    # active frame in v10. Save proof crops through the normal
                    # max-100/cooldown mechanism.
                    if (
                        ui.get("fish") is not None
                        and ui.get("fish_score", 0.0)
                        >= FISH_ACTIVE_MIN_SCORE
                    ):
                        local_fish_for_save = (
                            ui["fish"][0] - crop_origin[0],
                            ui["fish"][1] - crop_origin[1],
                            ui["fish"][2],
                            ui["fish"][3],
                        )

                        save_reference_hit(
                            fast_frame,
                            ACTIVE_MONITOR,
                            (
                                "fish-left.png"
                                if ui.get("fish_direction") == "left"
                                else "fish-right.png"
                            ),
                            ui.get("fish_score", 0.0),
                            local_fish_for_save,
                            FISH_ACTIVE_MIN_SCORE,
                        )

                    if ui.get("zone") is not None:
                        local_green_for_save = (
                            ui["zone"][0] - crop_origin[0],
                            ui["zone"][1] - crop_origin[1],
                            ui["zone"][2],
                            ui["zone"][3],
                        )

                        save_reference_hit(
                            fast_frame,
                            ACTIVE_MONITOR,
                            "green-target.png",
                            1.0,
                            local_green_for_save,
                            0.99,
                        )

                    if (
                        cached_ready_box is not None
                        and cached_ready_score
                        >= READY_THRESHOLD
                    ):
                        # Convert full-monitor ready box back to local crop for
                        # the exact hit image, avoiding a costly full-screen grab.
                        local_ready_for_save = (
                            cached_ready_box[0] - crop_origin[0],
                            cached_ready_box[1] - crop_origin[1],
                            cached_ready_box[2],
                            cached_ready_box[3],
                        )

                        save_reference_hit(
                            fast_frame,
                            ACTIVE_MONITOR,
                            cached_ready_ref
                            or "ready_space.png",
                            cached_ready_score,
                            local_ready_for_save,
                            READY_THRESHOLD,
                        )

                    # -----------------------------------------------------
                    # Console tracker telemetry
                    # -----------------------------------------------------
                    tracker_cycles += 1

                    if (
                        now - last_tracker_fps_log
                        >= 1.0
                    ):
                        elapsed_tracker = (
                            time.perf_counter()
                            - tracker_fps_started
                        )

                        measured_fps = (
                            tracker_cycles
                            / max(
                                elapsed_tracker,
                                0.001,
                            )
                        )

                        print(
                            f"[TRACK] round={round_number} "
                            f"M{ACTIVE_MONITOR} "
                            f"{measured_fps:.1f} scans/s "
                            f"fish={fish_x:.1f} "
                            f"green={green_x:.1f} "
                            f"lead={lead:+.1f} "
                            f"err={error:+.1f} "
                            f"key={desired} "
                            f"greenMode={ui.get('green_mode', '?')}",
                            flush=True,
                        )

                        tracker_cycles = 0
                        tracker_fps_started = (
                            time.perf_counter()
                        )
                        last_tracker_fps_log = now

                    if (
                        now - last_fish_log_at
                        >= FISH_LOG_INTERVAL
                    ):
                        print(
                            f"[FISH] round={round_number} "
                            f"fishX={fish_x:.1f} "
                            f"greenX={green_x:.1f} "
                            f"error={error:+.1f}px "
                            f"key={desired} "
                            f"greenMode={ui.get('green_mode', '?')} "
                            f"fishSource={ui.get('fish_source', '?')} "
                            f"inside={'YES' if raw_inside else 'NO'} "
                            f"progress={cached_progress*100:.1f}% "
                            f"ready={cached_ready_score:.3f}/{READY_THRESHOLD:.2f}",
                            flush=True,
                        )

                        last_fish_log_at = now

                    # -----------------------------------------------------
                    # Catch - v11 delayed SPACE while tracking continues
                    # -----------------------------------------------------
                    #
                    # IMPORTANT:
                    # We do NOT release A/D when the ready prompt first appears.
                    # The green target keeps following the fish throughout the
                    # complete safety delay.
                    #
                    ready_visible = (
                        cached_ready_score
                        >= READY_THRESHOLD
                    )

                    if ready_visible:
                        if ready_visible_since is None:
                            ready_visible_since = time.perf_counter()

                            print(
                                f"[READY] Round {round_number}: "
                                f"prompt appeared -> starting "
                                f"{SPACE_READY_DELAY_SECONDS*1000:.0f}ms "
                                f"SPACE safety delay; A/D TRACKING CONTINUES",
                                flush=True,
                            )
                    else:
                        if ready_visible_since is not None:
                            print(
                                f"[READY] Round {round_number}: "
                                "prompt disappeared -> delay reset",
                                flush=True,
                            )

                        ready_visible_since = None

                    ready_elapsed = (
                        time.perf_counter() - ready_visible_since
                        if ready_visible_since is not None
                        else 0.0
                    )

                    if (
                        ready_visible
                        and ready_elapsed
                        >= SPACE_READY_DELAY_SECONDS
                        and raw_inside
                    ):
                        # Only NOW stop movement and press SPACE. Until this
                        # exact point the normal A/D controller above has kept
                        # the green target following the fish.
                        release_movement()

                        dbg_event(
                            "catch",
                            round=round_number,
                            monitor=ACTIVE_MONITOR,
                            fish_x=fish_x,
                            green_x=green_x,
                            green_w=zone_width,
                            error=error,
                            desired_key=desired,
                            ready_elapsed_ms=ready_elapsed * 1000.0,
                        )

                        dbg_freeze(
                            "catch",
                            round=round_number,
                            ready_elapsed_ms=ready_elapsed * 1000.0,
                        )

                        print(
                            f"[CATCH] Round {round_number}: "
                            f"ready visible for "
                            f"{ready_elapsed*1000:.0f}ms "
                            f"+ fish in green -> SPACE",
                            flush=True,
                        )

                        tap("SPACE")

                        reset_tracker()
                        state = "WAIT_CAST"
                        post_catch_started_at = now

                        # Permit the next cast immediately when cast.png appears.
                        last_cast_at = (
                            now
                            - CAST_DEBOUNCE_SECONDS
                            - 0.1
                        )

                        print(
                            "[STATE] -> WAIT_CAST "
                            "(next cast.png will trigger E immediately)",
                            flush=True,
                        )

                    elif (
                        ready_visible
                        and ready_visible_since is not None
                        and now - last_fish_log_at < 0.03
                    ):
                        # Compact telemetry so the console proves it is waiting
                        # deliberately rather than being stuck.
                        print(
                            f"[READY] delay="
                            f"{ready_elapsed*1000:.0f}/"
                            f"{SPACE_READY_DELAY_SECONDS*1000:.0f}ms "
                            f"inside={'YES' if raw_inside else 'NO'} "
                            f"trackingKey={desired}",
                            flush=True,
                        )

                    # Low-rate full monitor debug only.
                    if (
                        ACTIVE_FULL_DEBUG
                        and now - last_debug_at
                        >= DEBUG_SAVE_INTERVAL
                    ):
                        full_debug = grab_monitor(
                            sct,
                            ACTIVE_MONITOR,
                        )

                        if full_debug is not None:
                            debug_ui = dict(ui)
                            debug_ui["valid"] = True
                            debug_ui["progress"] = cached_progress
                            debug_ui["progress_box"] = cached_progress_box
                            debug_ui["ready_score"] = cached_ready_score
                            debug_ui["ready_box"] = cached_ready_box
                            debug_ui["predicted_fish_x"] = predicted_fish_x

                            save_debug(
                                full_debug,
                                state,
                                debug_ui,
                                last_scan.get(
                                    ACTIVE_MONITOR,
                                    {
                                        "waiting_score": 0.0,
                                        "waiting_box": None,
                                        "bite_score": 0.0,
                                        "bite_box": None,
                                        "cast_score": 0.0,
                                        "cast_box": None,
                                    },
                                ),
                                ACTIVE_MONITOR,
                            )

                        last_debug_at = now

                # =========================================================
                # SEARCH / ARMED / WAIT_CAST
                # =========================================================
                else:
                    release_movement()

                    if (
                        state == "ARMED_FAST"
                        and ARMED_SKIP_FULL_SCAN
                        and now - armed_fast_started_at
                        < ARMED_SKIP_FULL_SCAN_SECONDS
                    ):
                        pass

                    elif (
                        now - last_scan_at
                        >= SCAN_INTERVAL
                    ):
                        last_scan_at = now

                        ordered = list(monitor_indices)

                        if ACTIVE_MONITOR in ordered:
                            ordered.remove(ACTIVE_MONITOR)
                            ordered.insert(
                                0,
                                ACTIVE_MONITOR,
                            )

                        for mi in ordered:
                            frame = grab_monitor(
                                sct,
                                mi,
                            )

                            if frame is None:
                                continue

                            # ------------------------------------------------
                            # After a completed catch, ONLY cast.png matters.
                            # This prevents stale previous-round minigame pixels
                            # from being treated as a new round.
                            # ------------------------------------------------
                            if state == "WAIT_CAST":
                                if (
                                    now - post_catch_started_at
                                    < POST_CATCH_MIN_DELAY
                                ):
                                    continue

                                (
                                    cast_score,
                                    cast_box,
                                    cast_scale,
                                ) = detect_cast(frame)

                                if (
                                    cast_score
                                    >= CAST_THRESHOLD
                                    and (
                                        now - last_cast_at
                                        >= CAST_DEBOUNCE_SECONDS
                                    )
                                ):
                                    save_reference_hit(
                                        frame,
                                        mi,
                                        "cast.png",
                                        cast_score,
                                        cast_box,
                                        CAST_THRESHOLD,
                                    )

                                    start_new_cast(
                                        sct,
                                        mi,
                                        cast_box,
                                        cast_score,
                                        cast_scale,
                                    )

                                    break

                                if (
                                    now - last_status_at
                                    >= STATUS_LOG_INTERVAL
                                ):
                                    print(
                                        f"[WAIT_CAST] M{mi} "
                                        f"cast={cast_score:.3f}/"
                                        f"{CAST_THRESHOLD:.2f}",
                                        flush=True,
                                    )

                                continue

                            # ------------------------------------------------
                            # IDLE/ARMED: acquire fresh minigame.
                            # ------------------------------------------------
                            ui = detect_minigame(frame)

                            if (
                                ui.get("valid")
                                and state == "IDLE"
                            ):
                                ACTIVE_MONITOR = mi

                                target_window_from_detection(
                                    sct,
                                    mi,
                                    ui["bar"],
                                )

                                reset_tracker()

                                locked_bar = ui["bar"]
                                last_known_bar_by_monitor[mi] = locked_bar
                                last_valid_minigame_at = now

                                print(
                                    f"[STATE] Round {round_number}: "
                                    f"-> FISHING on M{mi}",
                                    flush=True,
                                )

                                print(
                                    f"[DETECT] initial fish refs: "
                                    f"L={ui['fish_left_score']:.3f} "
                                    f"R={ui['fish_right_score']:.3f} "
                                    f"chosen={ui['fish_direction']}",
                                    flush=True,
                                )

                                state = "FISHING"

                                break

                            # ------------------------------------------------
                            # Text references / cast
                            # ------------------------------------------------
                            (
                                waiting_score,
                                waiting_box,
                                waiting_scale,
                            ) = detect_waiting(frame)

                            (
                                bite_score,
                                bite_box,
                                bite_scale,
                            ) = detect_bite(frame)

                            (
                                cast_score,
                                cast_box,
                                cast_scale,
                            ) = detect_cast(frame)

                            scan = {
                                "waiting_score": waiting_score,
                                "waiting_box": waiting_box,
                                "waiting_scale": waiting_scale,
                                "bite_score": bite_score,
                                "bite_box": bite_box,
                                "bite_scale": bite_scale,
                                "cast_score": cast_score,
                                "cast_box": cast_box,
                                "cast_scale": cast_scale,
                            }

                            last_scan[mi] = scan

                            save_reference_hit(
                                frame,
                                mi,
                                "waiting.png",
                                waiting_score,
                                waiting_box,
                                WAITING_THRESHOLD,
                            )

                            save_reference_hit(
                                frame,
                                mi,
                                "bite.png",
                                bite_score,
                                bite_box,
                                BITE_THRESHOLD,
                            )

                            save_reference_hit(
                                frame,
                                mi,
                                "cast.png",
                                cast_score,
                                cast_box,
                                CAST_THRESHOLD,
                            )

                            if (
                                cast_score
                                >= CAST_THRESHOLD
                                and (
                                    now - last_cast_at
                                    >= CAST_DEBOUNCE_SECONDS
                                )
                            ):
                                start_new_cast(
                                    sct,
                                    mi,
                                    cast_box,
                                    cast_score,
                                    cast_scale,
                                )
                                break

                            if (
                                now - last_status_at
                                >= STATUS_LOG_INTERVAL
                            ):
                                print(
                                    f"[SCAN] M{mi} state={state} "
                                    f"wait={waiting_score:.3f} "
                                    f"bite={bite_score:.3f} "
                                    f"cast={cast_score:.3f}/"
                                    f"{CAST_THRESHOLD:.2f} "
                                    f"bar={'YES' if ui.get('bar') else 'NO'} "
                                    f"zone={'YES' if ui.get('zone') else 'NO'}",
                                    flush=True,
                                )

                        if (
                            now - last_status_at
                            >= STATUS_LOG_INTERVAL
                        ):
                            last_status_at = now

                # =========================================================
                # Loop limiting
                # =========================================================
                elapsed = (
                    time.perf_counter()
                    - loop_started
                )

                # Active FISHING is deliberately uncapped.
                if not (
                    (
                        state == "FISHING"
                        and TRACK_UNCAPPED
                    )
                    or (
                        state == "ARMED_FAST"
                        and FAST_ARMED_UNCAPPED
                    )
                ):
                    wait = (
                        1.0 / TARGET_FPS
                        - elapsed
                    )

                    if wait > 0:
                        time.sleep(wait)

        except KeyboardInterrupt:
            print("\n[STATE] Ctrl+C -> EXIT", flush=True)

        finally:
            release_movement()

            dbg_event(
                "program_exit",
                round=round_number,
                state=state,
                monitor=ACTIVE_MONITOR,
            )

            dbg_freeze(
                "program_exit",
                round=round_number,
                state=state,
            )

            debug_zip = dbg_finalize()

            if debug_zip is not None:
                print(
                    "[IMPORTANT] SEND THIS ZIP BACK TO CHATGPT:",
                    flush=True,
                )
                print(
                    f"[IMPORTANT] {debug_zip}",
                    flush=True,
                )

            print("[DONE] A/D released.", flush=True)


if __name__ == "__main__":
    main()
