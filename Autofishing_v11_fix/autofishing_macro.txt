# AUTOFISHING v11
# Windows screen-only singleplayer fishing automation - v11
#
# This version uses the exact reference images in ./references/
# supplied by the user.
#
# F8 = pause/resume
# F9 = quit
#
# Console:
#   [REF]    loaded reference image
#   [WINDOW] game-window detection
#   [SCAN]   current reference/detection scores
#   [STATE]  state transitions
#   [FISH]   fish / target / progress values
#   [KEY]    every actual A/D/E/SPACE input
#   [DEBUG]  screenshot output
#
# IMPORTANT v5 DESIGN:
# - Fish-left.png and fish-right.png are NEVER searched over the whole screen.
# - First the unique long cyan fishing bar must be found.
# - Only INSIDE that bar are fish-left/fish-right searched.
# - A valid minigame requires:
#       cyan main bar
#       + one of the supplied fish references inside it
#       + green target zone inside the same bar
# - This prevents unrelated blue icons elsewhere on the screen from becoming "fish".

        subprocess.check_call(

        print("Run manually:", flush=True)
        print("  py -m pip install --user numpy opencv-python mss", flush=True)
        input("Press Enter to close...")
        raise SystemExit(1)

    os.execv(sys.executable, [sys.executable] + sys.argv)

# v7 still does NOT require any specific FiveM/GTA window title.
# It scans every physical monitor directly.

# the window physically UNDER that detected UI and remembers/focuses it before

if os.name != "nt":
    print("[ERROR] This script is intended for Windows.", flush=True)
    raise SystemExit(1)

user32 = ctypes.windll.user32

try:
    user32.SetProcessDPIAware()
except Exception:
    pass

ACTIVE_MONITOR = None
TARGET_HWND = None

GA_ROOT = 2


class POINT(ctypes.Structure):
    _fields_ = [
        ("x", wintypes.LONG),
        ("y", wintypes.LONG),
    ]


def hwnd_title(hwnd):
    if not hwnd:
        return ""

    length = user32.GetWindowTextLengthW(hwnd)

    if length <= 0:
        return ""

    buf = ctypes.create_unicode_buffer(length + 1)
    user32.GetWindowTextW(hwnd, buf, length + 1)
    return buf.value


def foreground_title():
    hwnd = user32.GetForegroundWindow()
    return hwnd_title(hwnd) or f"HWND {hwnd}"


def target_window_from_detection(sct, monitor_index, box):
    global TARGET_HWND

    if box is None:
        return None

    mon = sct.monitors[monitor_index]

    x, y, w, h = box

    screen_x = int(mon["left"] + x + w / 2)
    screen_y = int(mon["top"] + y + h / 2)

    pt = POINT(screen_x, screen_y)

    hwnd = user32.WindowFromPoint(pt)

    if hwnd:
        root = user32.GetAncestor(hwnd, GA_ROOT)

        if root:
            hwnd = root

    if hwnd:
        changed = hwnd != TARGET_HWND
        TARGET_HWND = hwnd

        if changed:
            print(
                f"[TARGET] Locked window under detection: "
                f"{hwnd_title(hwnd) or ('HWND ' + str(hwnd))}",
                flush=True,
            )

    return hwnd


def focus_target():
    if not TARGET_HWND:
        return False

    try:
        user32.SetForegroundWindow(TARGET_HWND)
        return True
    except Exception:
        return False


def focus_game():
    # Kept as the shared input hook name.
    focus_target()

INPUT_KEYBOARD = 1
KEYEVENTF_SCANCODE = 0x0008
KEYEVENTF_KEYUP = 0x0002

VK_F8 = 0x77
VK_F9 = 0x78

VK_CODE = {
    "A": 0x41,
    "D": 0x44,
    "E": 0x45,
    "SPACE": 0x20,
}

SCAN_CODE = {
    "A": 0x1E,
    "D": 0x20,
    "E": 0x12,
    "SPACE": 0x39,
}

ULONG_PTR = wintypes.WPARAM


class KEYBDINPUT(ctypes.Structure):
    _fields_ = [
        ("wVk", wintypes.WORD),
        ("wScan", wintypes.WORD),
        ("dwFlags", wintypes.DWORD),
        ("time", wintypes.DWORD),
        ("dwExtraInfo", ULONG_PTR),
    ]


class INPUTUNION(ctypes.Union):
    _fields_ = [("ki", KEYBDINPUT)]


class INPUT(ctypes.Structure):
    _anonymous_ = ("u",)
    _fields_ = [
        ("type", wintypes.DWORD),
        ("u", INPUTUNION),
    ]


held = {
    "A": False,
    "D": False,
}


def send_input_event(key, down):
    """
    Stronger v7 keyboard injection.

    Sends the physical scan-code through SendInput AND a matching keybd_event.
    This improves compatibility with games that ignore one of the two Windows
    keyboard paths.

    Console reports whether SendInput itself accepted the event.
    """
    focus_target()

    flags = KEYEVENTF_SCANCODE

    if not down:
        flags |= KEYEVENTF_KEYUP

    inp = INPUT(
        type=INPUT_KEYBOARD,
        ki=KEYBDINPUT(
            wVk=0,
            wScan=SCAN_CODE[key],
            dwFlags=flags,
            time=0,
            dwExtraInfo=0,
        ),
    )

    sent = user32.SendInput(
        1,
        ctypes.byref(inp),
        ctypes.sizeof(INPUT),
    )

    # Compatibility path.
    kb_flags = 0 if down else KEYEVENTF_KEYUP
    user32.keybd_event(
        VK_CODE[key],
        SCAN_CODE[key],
        kb_flags,
        0,
    )

    print(
        f"[INPUT] {key} {'DOWN' if down else 'UP'} "
        f"SendInput={sent}/1 "
        f"foreground='{foreground_title()}'",
        flush=True,
    )

    return sent == 1


def key_down(key):
    if key in held and held[key]:
        return

    send_input_event(key, True)

    if key in held:
        held[key] = True

    print(f"[KEY] DOWN {key}", flush=True)


def key_up(key):
    if key in held and not held[key]:
        return

    send_input_event(key, False)

    if key in held:
        held[key] = False

    print(f"[KEY] UP   {key}", flush=True)


def tap(key, duration=0.080):
    print(f"[KEY] TAP  {key}", flush=True)

    send_input_event(key, True)
    time.sleep(duration)
    send_input_event(key, False)


def release_movement():
    key_up("A")
    key_up("D")


def drive(direction):
    if direction == "A":
        if held["D"]:
            key_up("D")

        key_down("A")

    elif direction == "D":
        if held["A"]:
            key_up("A")

        key_down("D")

    else:
        release_movement()


def pressed_once(vk, old_state):
    state = bool(user32.GetAsyncKeyState(vk) & 0x8000)
    return state and not old_state, state

def grab_monitor(sct, monitor_index):
    try:
        shot = sct.grab(sct.monitors[monitor_index])
    except Exception:
        return None

    return np.asarray(shot, dtype=np.uint8)[:, :, :3].copy()


def grab_locked_fishing_region(sct, monitor_index, bar):
    """
    Capture only the fishing UI region instead of the entire monitor.

    Returns:
      (frame, local_bar, crop_origin_xy)

    local_bar uses coordinates inside the returned frame.
    crop_origin_xy is the offset back to monitor coordinates.
    """
    if bar is None:
        return None, None, None

    mon = sct.monitors[monitor_index]
    bx, by, bw, bh = map(int, bar)

    x0 = max(0, bx - FAST_REGION_SIDE_PADDING)
    y0 = max(0, by - FAST_REGION_TOP_PADDING)

    x1 = min(
        int(mon["width"]),
        bx + bw + FAST_REGION_SIDE_PADDING,
    )
    y1 = min(
        int(mon["height"]),
        by + bh + FAST_REGION_BOTTOM_PADDING,
    )

    if x1 <= x0 or y1 <= y0:
        return None, None, None

    region = {
        "left": int(mon["left"] + x0),
        "top": int(mon["top"] + y0),
        "width": int(x1 - x0),
        "height": int(y1 - y0),
    }

    try:
        shot = sct.grab(region)
    except Exception:
        return None, None, None

    frame = np.asarray(
        shot,
        dtype=np.uint8,
    )[:, :, :3].copy()

    local_bar = (
        bx - x0,
        by - y0,
        bw,
        bh,
    )

    return frame, local_bar, (x0, y0)

def prepared_gray(img):
    g = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    return cv2.GaussianBlur(g, (3, 3), 0)


REF_GRAY = {
    key: prepared_gray(img)
    for key, img in REF.items()
    if key != "minigame_example"
}


def multiscale_match(frame, ref_key, roi=None, scales=GENERAL_SCALES):
    H, W = frame.shape[:2]

    if roi is None:
        x0, y0, x1, y1 = 0, 0, W, H
    else:
        x0, y0, x1, y1 = map(int, roi)

        x0 = max(0, min(W, x0))
        x1 = max(0, min(W, x1))
        y0 = max(0, min(H, y0))
        y1 = max(0, min(H, y1))

    if x1 <= x0 or y1 <= y0:
        return 0.0, None, None

    search = prepared_gray(frame[y0:y1, x0:x1])
    templ0 = REF_GRAY[ref_key]

    th0, tw0 = templ0.shape

    best_score = -1.0
    best_box = None
    best_scale = None

    for scale in scales:
        tw = max(8, int(tw0 * scale))
        th = max(8, int(th0 * scale))

        if tw >= search.shape[1] or th >= search.shape[0]:
            continue

        interpolation = (
            cv2.INTER_AREA
            if scale < 1.0
            else cv2.INTER_LINEAR
        )

        templ = cv2.resize(
            templ0,
            (tw, th),
            interpolation=interpolation,
        )

        result = cv2.matchTemplate(
            search,
            templ,
            cv2.TM_CCOEFF_NORMED,
        )

        _, score, _, loc = cv2.minMaxLoc(result)

        if score > best_score:
            best_score = float(score)
            best_box = (
                int(loc[0] + x0),
                int(loc[1] + y0),
                int(tw),
                int(th),
            )
            best_scale = float(scale)

    return best_score, best_box, best_scale


def detect_waiting(frame):
    H, W = frame.shape[:2]

    # Notification is in the upper area.
    roi = (
        int(W * 0.08),
        0,
        int(W * 0.88),
        int(H * 0.34),
    )

    return multiscale_match(
        frame,
        "waiting",
        roi,
        GENERAL_SCALES,
    )

def detect_cast(frame):
    H, W = frame.shape[:2]

    # The supplied cast reference is searched only from the gameplay
    # interaction region, not from the entire screen.
    roi = (
        int(W * 0.52),

        int(H * 0.88),

def detect_ready(frame, bar):
    H, W = frame.shape[:2]
    bx, by, bw, bh = bar

    # The ready message is immediately above the main fishing bar.
    roi = (
        max(0, int(bx - bw * 0.28)),
        max(0, int(by - H * 0.18)),
        min(W, int(bx + bw * 1.28)),
        min(H, int(by + bh * 0.10)),
    )

    s1, b1, sc1 = multiscale_match(
        frame,
        "ready",
        roi,
        READY_SCALES,
    )

    s2, b2, sc2 = multiscale_match(
        frame,
        "ready_dark",
        roi,
        READY_SCALES,
    )

    if s1 >= s2:
        return s1, b1, sc1, "ready_space.png"

    return s2, b2, sc2, "ready_space-Darkerbackround.png"

def connected_components(mask, minimum_area=1):
    count, _, stats, centers = cv2.connectedComponentsWithStats(mask)

    result = []

    for index in range(1, count):
        x, y, w, h, area = stats[index]

        if area < minimum_area:
            continue

        cx, cy = centers[index]

        result.append(
            {
                "x": int(x),
                "y": int(y),
                "w": int(w),
                "h": int(h),
                "area": int(area),
                "cx": float(cx),
                "cy": float(cy),
            }
        )

    return result


def find_cyan_bar_candidates(frame):
    H, W = frame.shape[:2]

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Cyan used by the supplied fishing UI.
    cyan = cv2.inRange(
        hsv,
        (88, 105, 65),
        (122, 255, 255),
    )

    # Small close joins anti-aliased cyan border segments.
    kernel = np.ones((3, 3), np.uint8)
    cyan_closed = cv2.morphologyEx(
        cyan,
        cv2.MORPH_CLOSE,
        kernel,
        iterations=1,
    )

    parts = connected_components(cyan_closed, 40)

    candidates = []

    for c in parts:
        x = c["x"]
        y = c["y"]
        w = c["w"]
        h = c["h"]

        aspect = w / max(h, 1)
        density = c["area"] / max(w * h, 1)

        # Fishing UI lives in the lower portion.
        if y < H * 0.34:
            continue

        # Main bar is a substantial fraction of screen width.
        if w < W * 0.18:
            continue

        if w > W * 0.92:
            continue

        if h < max(18, H * 0.020):
            continue

        if h > H * 0.16:
            continue

        if aspect < 5.0:
            continue

        if aspect > 28.0:
            continue

        if density < 0.055:
            continue

        candidates.append(
            {
                "box": (x, y, w, h),
                "area": c["area"],
                "density": density,
                "aspect": aspect,
            }
        )

    # Wider candidates first; final validation uses fish reference + green target.
    candidates.sort(
        key=lambda c: (c["box"][2], c["area"]),
        reverse=True,
    )

    return candidates, cyan


def find_fish_in_bar(frame, bar):
    H, W = frame.shape[:2]
    bx, by, bw, bh = bar

    # Search tightly around/inside the cyan bar only.
    # A little vertical padding accounts for anti-aliasing/crop differences.
    roi = (
        max(0, bx),
        max(0, int(by - bh * 0.20)),
        min(W, bx + bw),
        min(H, int(by + bh * 1.20)),
    )

    left_score, left_box, left_scale = multiscale_match(
        frame,
        "fish_left",
        roi,
        FISH_SCALES,
    )

    right_score, right_box, right_scale = multiscale_match(
        frame,
        "fish_right",
        roi,
        FISH_SCALES,
    )

    if left_score >= right_score:
        return {
            "score": left_score,
            "box": left_box,
            "scale": left_scale,
            "direction": "left",
            "left_score": left_score,
            "right_score": right_score,
        }

    return {
        "score": right_score,
        "box": right_box,
        "scale": right_scale,
        "direction": "right",
        "left_score": left_score,
        "right_score": right_score,
    }


def find_green_zone(frame, bar):
    H, W = frame.shape[:2]
    bx, by, bw, bh = bar

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Bright UI green from the user's exact examples.
    green = cv2.inRange(
        hsv,
        (43, 115, 65),
        (93, 255, 255),
    )

    x0 = max(0, bx)
    x1 = min(W, bx + bw)

    y0 = max(0, int(by - bh * 0.10))
    y1 = min(H, int(by + bh * 1.10))

    crop = green[y0:y1, x0:x1]

    if crop.size == 0:
        return None

    # A vertical target may contain thin border lines plus translucent fill.
    # Closing joins the target into a stable region.
    kernel = np.ones((3, 3), np.uint8)
    crop = cv2.morphologyEx(
        crop,
        cv2.MORPH_CLOSE,
        kernel,
        iterations=1,
    )

    parts = connected_components(crop, 20)

    candidates = []

    for c in parts:
        w = c["w"]
        h = c["h"]
        area = c["area"]

        if h < bh * 0.48:
            continue

        if h > bh * 1.20:
            continue

        if w < 3:
            continue

        if w > bw * 0.22:
            continue

        # Target is vertical / roughly square-ish, never a long progress bar.
        if w / max(h, 1) > 1.55:
            continue

        gx = c["x"] + x0
        gy = c["y"] + y0

        candidates.append(
            {
                "box": (gx, gy, w, h),
                "area": area,
            }
        )

    if not candidates:
        return None

    # Prefer largest coherent green region within the main bar.
    candidates.sort(
        key=lambda c: c["area"],
        reverse=True,
    )

    return candidates[0]["box"]


def detect_progress(frame, bar):
    H, W = frame.shape[:2]
    bx, by, bw, bh = bar

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Lower progress is cyan -> green.
    fill = cv2.inRange(
        hsv,
        (34, 100, 55),
        (124, 255, 255),
    )

    x0 = max(0, int(bx - bw * 0.035))
    x1 = min(W, int(bx + bw * 1.035))

    y0 = max(0, int(by + bh + 2))
    y1 = min(H, int(by + bh + max(35, H * 0.11)))

    crop = fill[y0:y1, x0:x1]

    if crop.size == 0:
        return 0.0, None

    row_counts = (crop > 0).sum(axis=1)

    if not row_counts.size:
        return 0.0, None

    best_row = int(np.argmax(row_counts))

    if row_counts[best_row] < max(20, bw * 0.06):
        return 0.0, None

    slab_top = max(0, best_row - 5)
    slab_bottom = min(crop.shape[0], best_row + 6)
    slab = crop[slab_top:slab_bottom]

    ys, xs = np.nonzero(slab)

    if len(xs) < 20:
        return 0.0, None

    left = float(np.percentile(xs, 1)) + x0
    right = float(np.percentile(xs, 99)) + x0

    ratio = (right - bx) / max(float(bw), 1.0)
    ratio = max(0.0, min(1.05, ratio))

    progress_box = (
        int(left),
        int(y0 + slab_top),
        int(max(1.0, right - left)),
        int(max(1, slab_bottom - slab_top)),
    )

    return ratio, progress_box


def remap_box(box, origin):
    if box is None:
        return None

    ox, oy = origin
    x, y, w, h = box

    return (
        int(x + ox),
        int(y + oy),
        int(w),
        int(h),
    )


def track_green_target_borders(
    frame,
    local_bar,
    previous_zone=None,
):
    """
    v10 robust green-target tracker.

    The user's empty green reference shows that the target is essentially one
    vertical green rectangle. In the real minigame, when the cyan fish overlaps
    it, that rectangle may be visually split into two green pieces.

    So we track the OUTER green span:
      - unobstructed target -> one continuous green run
      - fish on target      -> two/three green runs whose OUTER edges still
                               describe the same target

    This avoids the v9 bug where one fragment could be mistaken for the whole
    target and move the calculated center to the wrong side.
    """
    H, W = frame.shape[:2]
    bx, by, bw, bh = map(int, local_bar)

    if bw <= 20 or bh <= 10:
        return None

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

    # Include both bright borders and the darker/translucent green interior.
    green = cv2.inRange(
        hsv,
        (35, 75, 50),
        (100, 255, 255),
    )

    x0 = max(0, bx + 1)
    x1 = min(W, bx + bw - 1)
    y0 = max(0, by + 1)
    y1 = min(H, by + bh - 1)

    crop = green[y0:y1, x0:x1]

    if crop.size == 0:
        return None

    inner_h = crop.shape[0]

    # A real target column is green through a meaningful part of the bar height.
    counts = (crop > 0).sum(axis=0).astype(np.float32)

    minimum_count = max(
        3.0,
        inner_h * 0.22,
    )

    active_cols = np.where(
        counts >= minimum_count
    )[0]

    if len(active_cols) == 0:
        return None

    # Build green runs. Small 1-2px holes are bridged because anti-aliasing /
    # fish edges can cut tiny gaps into the target.
    runs = []
    start_col = prev_col = int(active_cols[0])

    for value in active_cols[1:]:
        value = int(value)

        if value <= prev_col + 3:
            prev_col = value
        else:
            runs.append(
                (
                    start_col,
                    prev_col,
                )
            )
            start_col = prev_col = value

    runs.append(
        (
            start_col,
            prev_col,
        )
    )

    expected_width = max(
        8.0,
        (bh - 2.0)
        * GREEN_REFERENCE_WIDTH_RATIO,
    )

    previous_center = None
    previous_width = None

    if previous_zone is not None:
        px, py, pw, ph = previous_zone

        previous_center = (
            px + pw / 2.0
        )

        previous_width = float(pw)

        if previous_width >= 6.0:
            expected_width = (
                0.82 * previous_width
                + 0.18 * expected_width
            )

    min_width = max(
        5.0,
        expected_width
        * GREEN_WIDTH_MIN_FACTOR,
    )

    max_width = max(
        min_width + 1.0,
        expected_width
        * GREEN_WIDTH_MAX_FACTOR,
    )

    candidates = []

    # Consider every contiguous sequence of nearby green runs.
    # This naturally handles:
    #   [ GREEN GREEN GREEN ]
    # and
    #   [ GREEN ][cyan fish][ GREEN ]
    for i in range(len(runs)):
        for j in range(i, len(runs)):
            left_local = runs[i][0]
            right_local = runs[j][1]

            span_width = (
                right_local
                - left_local
                + 1
            )

            if span_width < min_width:
                continue

            if span_width > max_width:
                # Later j values can only become wider.
                break

            # Don't combine completely unrelated green UI pieces separated by a
            # huge gap. Fish occlusion can make a gap, but the total span still
            # needs to look like the supplied target.
            covered_columns = 0

            for k in range(i, j + 1):
                covered_columns += (
                    runs[k][1]
                    - runs[k][0]
                    + 1
                )

            coverage_ratio = (
                covered_columns
                / max(
                    float(span_width),
                    1.0,
                )
            )

            # If the fish is sitting over the target, coverage can drop.
            # Keep this permissive but reject very sparse unrelated pairs.
            if coverage_ratio < 0.28:
                continue

            left_x = (
                x0 + left_local
            )

            right_x = (
                x0 + right_local
            )

            center = (
                left_x + right_x
            ) / 2.0

            width_error = abs(
                span_width
                - expected_width
            ) / max(
                expected_width,
                1.0,
            )

            continuity_penalty = 0.0

            if previous_center is not None:
                jump = abs(
                    center
                    - previous_center
                )

                if (
                    jump
                    > expected_width
                    * GREEN_TRACK_MAX_JUMP_FACTOR
                ):
                    continue

                continuity_penalty = (
                    jump
                    / max(
                        expected_width,
                        1.0,
                    )
                    * 0.45
                )

            # Prefer spans close to expected width, close to last center, and
            # with more actual green coverage.
            score = (
                width_error
                + continuity_penalty
                + (1.0 - coverage_ratio)
                * 0.12
            )

            candidates.append(
                (
                    score,
                    left_x,
                    right_x,
                    center,
                    float(span_width),
                    coverage_ratio,
                    i,
                    j,
                )
            )

    if candidates:
        candidates.sort(
            key=lambda item: item[0],
        )

        (
            _,
            left_x,
            right_x,
            center,
            width,
            coverage_ratio,
            run_i,
            run_j,
        ) = candidates[0]

        mode = (
            "full-span"
            if run_i == run_j
            else "split-span"
        )

        return {
            "box": (
                int(round(left_x)),
                int(y0),
                max(
                    1,
                    int(round(
                        right_x
                        - left_x
                        + 1
                    )),
                ),
                int(y1 - y0),
            ),
            "center_x": float(center),
            "width": float(width),
            "mode": mode,
            "coverage": float(
                coverage_ratio
            ),
        }

    # Last-resort continuity fallback:
    # if only a narrow fragment is visible for one frame, infer which edge it
    # belongs to from the previous target center/width.
    if (
        previous_center is not None
        and previous_width is not None
        and runs
    ):
        best_inferred = None

        for r0, r1 in runs:
            run_center = (
                x0
                + (r0 + r1) / 2.0
            )

            # Interpret the visible run as either left side or right side.
            centers = (
                run_center
                + previous_width / 2.0,
                run_center
                - previous_width / 2.0,
            )

            for inferred_center in centers:
                jump = abs(
                    inferred_center
                    - previous_center
                )

                if (
                    jump
                    > expected_width
                    * GREEN_TRACK_MAX_JUMP_FACTOR
                ):
                    continue

                if (
                    best_inferred is None
                    or jump
                    < best_inferred[0]
                ):
                    best_inferred = (
                        jump,
                        inferred_center,
                    )

        if best_inferred is not None:
            _, center = best_inferred

            return {
                "box": (
                    int(round(
                        center
                        - previous_width / 2.0
                    )),
                    int(y0),
                    max(
                        1,
                        int(round(
                            previous_width
                        )),
                    ),
                    int(y1 - y0),
                ),
                "center_x": float(center),
                "width": float(
                    previous_width
                ),
                "mode": "fragment-inferred",
                "coverage": 0.0,
            }

    return None



def fast_match_fish_in_locked_bar(
    frame,
    local_bar,
    previous_scale=None,
):
    """
    Exact fish-left/fish-right template matching every active tracker frame,
    but only inside the tiny known bar and at a narrow scale range.
    """
    H, W = frame.shape[:2]
    bx, by, bw, bh = map(int, local_bar)

    roi = (
        max(0, bx),
        max(0, int(by - bh * 0.10)),
        min(W, bx + bw),
        min(H, int(by + bh * 1.10)),
    )

    if previous_scale is None:
        # Calibrated default from the user's examples.
        base_scale = 0.54
    else:
        base_scale = float(previous_scale)

    scales = tuple(
        max(
            0.28,
            min(
                1.40,
                base_scale * factor,
            ),
        )
        for factor in FISH_ACTIVE_SCALE_FACTORS
    )

    left_score, left_box, left_scale = multiscale_match(
        frame,
        "fish_left",
        roi,
        scales,
    )

    right_score, right_box, right_scale = multiscale_match(
        frame,
        "fish_right",
        roi,
        scales,
    )

    if left_score >= right_score:
        return {
            "score": float(left_score),
            "box": left_box,
            "scale": left_scale,
            "direction": "left",
            "left_score": float(left_score),
            "right_score": float(right_score),
        }

    return {
        "score": float(right_score),
        "box": right_box,
        "scale": right_scale,
        "direction": "right",
        "left_score": float(left_score),
        "right_score": float(right_score),
    }


def fast_track_minigame(frame, local_bar, previous_ui=None):
    """
    v10 active tracker.

    Fish:
      exact user-supplied fish-left / fish-right template every frame.

    Green target:
      left + right bright-green border pair every frame.

    Neither center is inferred from a fragmented color blob anymore.
    """
    if frame is None or local_bar is None:
        return {
            "valid": False,
            "bar": local_bar,
            "fish": None,
            "zone": None,
            "fish_score": 0.0,
            "fish_left_score": 0.0,
            "fish_right_score": 0.0,
            "fish_direction": None,
            "fish_scale": None,
        }

    previous_scale = None
    previous_zone = None

    if previous_ui:
        previous_scale = previous_ui.get(
            "fish_scale"
        )
        previous_zone = previous_ui.get(
            "zone"
        )

    fish = fast_match_fish_in_locked_bar(
        frame,
        local_bar,
        previous_scale=previous_scale,
    )

    green = track_green_target_borders(
        frame,
        local_bar,
        previous_zone=previous_zone,
    )

    if (
        fish["box"] is None
        or fish["score"] < FISH_ACTIVE_MIN_SCORE
        or green is None
    ):
        return {
            "valid": False,
            "bar": local_bar,
            "fish": fish["box"],
            "zone": (
                green["box"]
                if green is not None
                else None
            ),
            "fish_score": float(fish["score"]),
            "fish_left_score": float(fish["left_score"]),
            "fish_right_score": float(fish["right_score"]),
            "fish_direction": fish["direction"],
            "fish_scale": fish["scale"],
            "green_mode": (
                green["mode"]
                if green is not None
                else None
            ),
        }

    fx, fy, fw, fh = fish["box"]
    fish_center_x = fx + fw / 2.0

    zone_box = green["box"]
    zone_center_x = green["center_x"]

    zx, zy, zw, zh = zone_box

    return {
        "valid": True,
        "bar": local_bar,
        "fish": fish["box"],
        "fish_score": float(fish["score"]),
        "fish_left_score": float(fish["left_score"]),
        "fish_right_score": float(fish["right_score"]),
        "fish_direction": fish["direction"],
        "fish_scale": fish["scale"],
        "zone": zone_box,
        "green_mode": green["mode"],
        "fish_center_x": float(fish_center_x),
        "zone_center_x": float(zone_center_x),
        "error": float(
            fish_center_x
            - zone_center_x
        ),
        "inside": bool(
            fish_center_x
            >= zx + zw * 0.04
            and fish_center_x
            <= zx + zw * 0.96
        ),
        "progress": 0.0,
        "progress_box": None,
        "ready_score": 0.0,
        "ready_box": None,
        "ready_scale": None,
        "ready_ref": None,
    }



def fish_blob_present(frame, local_bar):
    """
    Sub-millisecond "has the fish appeared inside the bar yet?" test.

    While ARMED_FAST waits for the bite there is only one question to answer,
    and stock v11 answered it by running the full fish-template match - two
    references at five scales, ~25 ms on a 1080p bar - on every loop.

    The fish icon is a compact BRIGHT cyan glyph. The bar's own fill carries a
    cyan gradient in the same hue range but it is dark: measured on the
    supplied minigame_example.png, the fish pixels have V 65..237 (median 237)
    while the bar gradient tops out at V 74. A brightness floor therefore
    separates them, and a blob of roughly the right size and shape means the
    fish is on screen.

    This is a GATE, not a detector. When it says yes, the normal v11 tracker
    runs on the same frame and produces the actual position, so nothing about
    detection accuracy changes. When it is unsure it says yes.
    """
    if frame is None or local_bar is None:
        return True

    H, W = frame.shape[:2]
    bx, by, bw, bh = map(int, local_bar)

    if bw <= 20 or bh <= 10:
        return True

    # Keep the bar's own bright border outside the crop.
    inset = max(3, int(round(bh * 0.07)))

    x0 = max(0, bx + inset)
    x1 = min(W, bx + bw - inset)
    y0 = max(0, by + inset)
    y1 = min(H, by + bh - inset)

    if x1 <= x0 or y1 <= y0:
        return True

    hsv = cv2.cvtColor(frame[y0:y1, x0:x1], cv2.COLOR_BGR2HSV)

    mask = cv2.inRange(hsv, (88, 105, 110), (123, 255, 255))

    count, _, stats, _ = cv2.connectedComponentsWithStats(mask)

    # Sizes relative to the bar height, so this works at any resolution.
    min_area = max(24.0, 0.05 * bh * bh)
    max_area = 1.60 * bh * bh
    max_width = bw * 0.35

    for index in range(1, count):
        x, y, w, h, area = stats[index]

        if area < min_area or area > max_area:
            continue

        if w > max_width:
            continue

        aspect = w / max(h, 1)

        if aspect < 0.30 or aspect > 3.20:
            continue

        return True

    return False


def detect_minigame(frame):
    candidates, cyan_mask = find_cyan_bar_candidates(frame)

    best_partial = None

    for candidate in candidates:
        bar = candidate["box"]

        fish = find_fish_in_bar(frame, bar)
        zone = find_green_zone(frame, bar)

        partial = {
            "bar": bar,
            "bar_aspect": candidate["aspect"],
            "bar_density": candidate["density"],
            "fish": fish["box"],
            "fish_score": fish["score"],
            "fish_left_score": fish["left_score"],
            "fish_right_score": fish["right_score"],
            "fish_direction": fish["direction"],
            "fish_scale": fish["scale"],
            "zone": zone,
            "valid": False,
        }

        if (
            best_partial is None
            or partial["fish_score"] > best_partial["fish_score"]
        ):
            best_partial = partial

        # A valid minigame MUST satisfy the supplied fish reference
        # and green target inside the same cyan bar.
        if fish["score"] < FISH_THRESHOLD:
            continue

        if fish["box"] is None:
            continue

        if zone is None:
            continue

        fx, fy, fw, fh = fish["box"]
        zx, zy, zw, zh = zone

        fish_center_x = fx + fw / 2.0
        zone_center_x = zx + zw / 2.0

        error = fish_center_x - zone_center_x

        # Give a small tolerance at the edges. The fish should be fully
        # controllable without requiring pixel-perfect containment.
        inside = (
            fish_center_x >= zx + zw * 0.04
            and fish_center_x <= zx + zw * 0.96
        )

        progress, progress_box = detect_progress(frame, bar)

        ready_score, ready_box, ready_scale, ready_ref = detect_ready(
            frame,
            bar,
        )

        return {
            "valid": True,
            "bar": bar,
            "fish": fish["box"],
            "fish_score": fish["score"],
            "fish_left_score": fish["left_score"],
            "fish_right_score": fish["right_score"],
            "fish_direction": fish["direction"],
            "fish_scale": fish["scale"],
            "zone": zone,
            "fish_center_x": float(fish_center_x),
            "zone_center_x": float(zone_center_x),
            "error": float(error),
            "inside": bool(inside),
            "progress": float(progress),
            "progress_box": progress_box,
            "ready_score": float(ready_score),
            "ready_box": ready_box,
            "ready_scale": ready_scale,
            "ready_ref": ready_ref,
        }

    if best_partial is not None:
        return best_partial

    return {
        "valid": False,
        "bar": None,
        "fish": None,
        "fish_score": 0.0,
        "fish_left_score": 0.0,
        "fish_right_score": 0.0,
        "fish_direction": None,
        "fish_scale": None,
        "zone": None,
    }

def draw_box(image, box, text):
    if box is None:
        return

    x, y, w, h = map(int, box)

    cv2.rectangle(
        image,
        (x, y),
        (x + w, y + h),
        (255, 255, 255),
        2,
    )

    cv2.putText(
        image,
        text,
        (x, max(22, y - 7)),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.52,
        (255, 255, 255),
        2,
        cv2.LINE_AA,
    )

        )

        if ui.get("valid"):
            # Visual proof of the actual controller inputs:
            # fish center, green center, and predicted fish position.
            fish_box = ui.get("fish")
            zone_box = ui.get("zone")

            if fish_box is not None and zone_box is not None:
                fx, fy, fw, fh = fish_box
                zx, zy, zw, zh = zone_box

                fish_cx = int(fx + fw / 2)
                fish_cy = int(fy + fh / 2)
                green_cx = int(zx + zw / 2)
                green_cy = int(zy + zh / 2)

                    ui.get(
                        "predicted_fish_x",

DETECTION_DIR = BASE / "detections"
DETECTION_DIR.mkdir(exist_ok=True)

_detection_sequence = 0
_last_detection_save = {}


def trim_detection_folder():
    files = sorted(
        DETECTION_DIR.glob("*.png"),
        key=lambda p: p.stat().st_mtime,
    )

    while len(files) > DETECTION_MAX_FILES:
        victim = files.pop(0)

        try:
            victim.unlink()
        except Exception:
            break


def save_reference_hit(
    frame,
    monitor_index,
    ref_name,
    score,
    box,
    threshold,
):
    """
    Save the EXACT matched rectangle, not the whole screenshot.

    A short cooldown prevents one static reference from filling all 100 files
    instantly while still producing a chronological detection trail.
    """
    global _detection_sequence

    if box is None or score < threshold:
        return

    now = time.monotonic()
    last = _last_detection_save.get((monitor_index, ref_name), 0.0)

    if now - last < DETECTION_SAVE_COOLDOWN:
        return

    _last_detection_save[(monitor_index, ref_name)] = now

    H, W = frame.shape[:2]
    x, y, w, h = map(int, box)

    x0 = max(0, x)
    y0 = max(0, y)
    x1 = min(W, x + w)
    y1 = min(H, y + h)

    if x1 <= x0 or y1 <= y0:
        return

    crop = frame[y0:y1, x0:x1].copy()

    _detection_sequence += 1

    safe_name = ref_name.replace("/", "_").replace("\\", "_")

    filename = (
        f"{_detection_sequence:04d}_M{monitor_index}_"
        f"{safe_name}_score-{score:.3f}.png"
    )

    path = DETECTION_DIR / filename
    cv2.imwrite(str(path), crop)

    trim_detection_folder()

    print(
        f"[SNAPSHOT] Saved exact hit crop: {filename} "
        f"box=({x},{y},{w},{h})",
        flush=True,
    )

def main():

    print("[INFO] After E, fresh fish/green acquisition is UNCAPPED and immediate.")
    print("[INFO] SPACE waits at least 220ms after the ready prompt appears.")
    print("[INFO] A/D tracking continues throughout the entire SPACE delay.")
    print("[INFO] Active fish tracking uses exact fish-left/right references every frame.")
    print("[INFO] Green target tracking uses its TWO bright border lines every frame.")
    print("[INFO] v10 uses DIRECT current positions: no prediction can send it the old way.")

    state = "IDLE"
    paused = False

    f8_old = False
    f9_old = False

    last_scan_at = 0.0
    last_status_at = 0.0
    last_fish_log_at = 0.0
    last_debug_at = 0.0
    last_cast_at = 0.0
    last_valid_minigame_at = 0.0

    post_catch_started_at = 0.0

    # Minimum SPACE delay state. Starts only when the ready-space reference is
    # actually visible and resets if the prompt disappears.
    ready_visible_since = None

    # Remember where the fishing bar was on each monitor across rounds.
    # This is intentionally NOT reset after a catch: the next cast normally
    # places the minigame in exactly the same location.
    last_known_bar_by_monitor = {}

    # Round/tracker state. These are hard-reset every time E starts a new cast.
    locked_bar = None
    fast_track_failures = 0

    # Responsiveness fix state.
    last_armed_full_reacquire = 0.0
    armed_gate_blocked_since = None
    armed_started_at = 0.0
    previous_fish_x = None
    previous_fish_time = None
    fish_velocity = 0.0
    previous_local_ui = None

    cached_progress = 0.0
    cached_progress_box = None
    cached_ready_score = 0.0
    cached_ready_box = None
    cached_ready_ref = None

    last_progress_check = 0.0
    last_ready_check = 0.0
    last_fish_reference_check = 0.0

    tracker_cycles = 0
    tracker_fps_started = time.perf_counter()
    last_tracker_fps_log = time.monotonic()

    round_number = 0

    last_scan = {}

    def reset_tracker():
        nonlocal locked_bar
        nonlocal fast_track_failures
        nonlocal previous_fish_x
        nonlocal previous_fish_time
        nonlocal fish_velocity
        nonlocal previous_local_ui
        nonlocal ready_visible_since
        nonlocal cached_progress
        nonlocal cached_progress_box
        nonlocal cached_ready_score
        nonlocal cached_ready_box
        nonlocal cached_ready_ref
        nonlocal last_progress_check
        nonlocal last_ready_check
        nonlocal last_fish_reference_check
        nonlocal tracker_cycles
        nonlocal tracker_fps_started
        nonlocal last_tracker_fps_log

        release_movement()

        locked_bar = None
        fast_track_failures = 0
        previous_fish_x = None
        previous_fish_time = None
        fish_velocity = 0.0
        previous_local_ui = None
        ready_visible_since = None

        cached_progress = 0.0
        cached_progress_box = None
        cached_ready_score = 0.0
        cached_ready_box = None
        cached_ready_ref = None

        last_progress_check = 0.0
        last_ready_check = 0.0
        last_fish_reference_check = 0.0

        tracker_cycles = 0
        tracker_fps_started = time.perf_counter()
        last_tracker_fps_log = time.monotonic()

    def start_new_cast(sct, monitor_index, cast_box, cast_score, cast_scale):
        nonlocal state
        nonlocal last_cast_at
        nonlocal round_number
        nonlocal locked_bar
        nonlocal fast_track_failures
        nonlocal last_armed_full_reacquire
        nonlocal armed_gate_blocked_since
        nonlocal armed_started_at
        global ACTIVE_MONITOR

        ACTIVE_MONITOR = monitor_index

        target_window_from_detection(
            sct,
            monitor_index,
            cast_box,
        )

        reset_tracker()

        round_number += 1

        print(
            f"[CAST] Round {round_number}: "
            f"cast.png on M{monitor_index} "
            f"score={cast_score:.3f} scale={cast_scale:.2f} -> E",
            flush=True,
        )

        tap("E")

        last_cast_at = time.monotonic()

        # Critical v11 behavior:
        # immediately start probing the previous fishing-bar position. No
        # SCAN_INTERVAL wait and no waiting/bite notification is required.
        locked_bar = last_known_bar_by_monitor.get(monitor_index)
        fast_track_failures = 0

        last_armed_full_reacquire = 0.0
        armed_gate_blocked_since = None
        armed_started_at = time.monotonic()

        state = "ARMED_FAST"

        if locked_bar is not None:
            print(
                f"[STATE] Round {round_number}: ARMED_FAST "
                f"(probing previous bar {locked_bar} immediately)",
                flush=True,
            )
        else:
            print(
                f"[STATE] Round {round_number}: ARMED_FAST "
                "(first round: searching for cyan bar immediately)",
                flush=True,
            )

    print("[STATE] IDLE", flush=True)

    with mss.mss() as sct:
        monitor_indices = list(range(1, len(sct.monitors)))

        if not monitor_indices:
            print("[ERROR] MSS found no physical monitors.", flush=True)
            input("Press Enter to close...")
            return

        print(
            f"[CAPTURE] Found {len(monitor_indices)} physical monitor(s).",
            flush=True,
        )

        for mi in monitor_indices:
            print(
                f"[CAPTURE]   {monitor_description(sct, mi)}",
                flush=True,
            )

        try:
            while True:
                loop_started = time.perf_counter()
                now = time.monotonic()

                f8_pressed, f8_old = pressed_once(VK_F8, f8_old)
                f9_pressed, f9_old = pressed_once(VK_F9, f9_old)

                if f9_pressed:
                    print("[STATE] F9 -> EXIT", flush=True)
                    break

                if f8_pressed:
                    paused = not paused
                    release_movement()

                    print(
                        "[STATE] PAUSED"
                        if paused
                        else "[STATE] RESUMED",
                        flush=True,
                    )

                if paused:
                    time.sleep(0.05)
                    continue

                # =========================================================
                # ARMED_FAST: after E, acquire the NEW round immediately
                # =========================================================
                #
                # The previous version returned to the normal screen-scanning
                # cadence after E. That could waste the first part of a new
                # minigame while the fish was already moving.
                #
                # v11 probes the known bar area as fast as possible. If the bar
                # position is not known yet, it does a cheap cyan-bar search on
                # the active monitor and locks it the instant it appears.
                #
                if (
                    state == "ARMED_FAST"
                    and ACTIVE_MONITOR in monitor_indices
                ):
                    armed_frame = None
                    local_bar = None
                    crop_origin = None
                    local_ui = None

                    # A bar is "confirmed" only when it is the one remembered
                    # from a round that actually produced a valid minigame.
                    # Anything else is a guess from a cyan blob that happened
                    # to look bar-shaped, and the shortcuts below must not
                    # trust it - otherwise the very first round throttles
                    # itself waiting on a bogus position.
                    armed_bar_confirmed = (
                        locked_bar is not None
                        and locked_bar
                        == last_known_bar_by_monitor.get(ACTIVE_MONITOR)
                    )

                    if locked_bar is not None:
                        (
                            armed_frame,
                            local_bar,
                            crop_origin,
                        ) = grab_locked_fishing_region(
                            sct,
                            ACTIVE_MONITOR,
                            locked_bar,
                        )

                        if armed_frame is not None:
                            # The fish template match is the expensive part of
                            # this loop (~25 ms). Run it only once the cheap
                            # colour gate says the fish is actually there, so
                            # the wait polls the bar hundreds of times a second
                            # instead of a dozen. The gate can never blind the
                            # script: after ARMED_GATE_SAFETY_INTERVAL of
                            # uninterrupted "no", the match runs regardless.
                            run_match = True

                            if ARMED_USE_FISH_GATE and armed_bar_confirmed:
                                if fish_blob_present(armed_frame, local_bar):
                                    armed_gate_blocked_since = None
                                else:
                                    if armed_gate_blocked_since is None:
                                        armed_gate_blocked_since = now

                                    run_match = (
                                        now - armed_gate_blocked_since
                                        >= ARMED_GATE_SAFETY_INTERVAL
                                    )

                                    if run_match:
                                        armed_gate_blocked_since = now

                            if run_match:
                                local_ui = fast_track_minigame(
                                    armed_frame,
                                    local_bar,
                                    previous_ui=None,
                                )

                    # If the previous bar location is unavailable or the UI
                    # layout changed, reacquire the cyan bar directly.
                    #
                    # This costs a full-monitor screenshot plus a whole-screen
                    # cyan search (~20 ms) and stock v11 paid it on EVERY loop
                    # of the wait, even though the bar does not move between
                    # rounds. It is now rate limited.
                    if (
                        (
                            local_ui is None
                            or not local_ui.get("valid")
                        )
                        and (
                            not armed_bar_confirmed
                            or now - last_armed_full_reacquire
                            >= ARMED_FULL_REACQUIRE_INTERVAL
                        )
                    ):
                        last_armed_full_reacquire = now

                        full_armed_frame = grab_monitor(
                            sct,
                            ACTIVE_MONITOR,
                        )

                        if full_armed_frame is not None:
                            bar_candidates, _ = find_cyan_bar_candidates(
                                full_armed_frame
                            )

                            if bar_candidates:
                                candidate_bar = bar_candidates[0]["box"]

                                if candidate_bar != locked_bar:
                                    locked_bar = candidate_bar

                                    print(
                                        f"[ARMED_FAST] Cyan bar appeared at "
                                        f"{locked_bar} on M{ACTIVE_MONITOR}",
                                        flush=True,
                                    )

                                (
                                    armed_frame,
                                    local_bar,
                                    crop_origin,
                                ) = grab_locked_fishing_region(
                                    sct,
                                    ACTIVE_MONITOR,
                                    locked_bar,
                                )

                                if armed_frame is not None:
                                    local_ui = fast_track_minigame(
                                        armed_frame,
                                        local_bar,
                                        previous_ui=None,
                                    )

                    if (
                        local_ui is not None
                        and local_ui.get("valid")
                    ):
                        # The fresh fish + fresh green target are visible NOW.
                        # Start the high-speed controller on the very next loop,
                        # with no normal scan delay in between.
                        last_known_bar_by_monitor[
                            ACTIVE_MONITOR
                        ] = locked_bar

                        last_valid_minigame_at = now
                        fast_track_failures = 0
                        previous_local_ui = dict(local_ui)

                        state = "FISHING"

                        print(
                            f"[STATE] Round {round_number}: "
                            f"ARMED_FAST -> FISHING IMMEDIATELY "
                            f"on M{ACTIVE_MONITOR}",
                            flush=True,
                        )

                        print(
                            f"[ARMED_FAST] Fresh fish/green acquired: "
                            f"fishScore={local_ui.get('fish_score', 0):.3f} "
                            f"greenMode={local_ui.get('green_mode', '?')}",
                            flush=True,
                        )

                        # No sleep at the end of this iteration because state
                        # is now FISHING; next loop starts immediately.

                # =========================================================
                # FISHING: locked-region high-speed color tracker
                # =========================================================
                if (
                    state == "FISHING"
                    and ACTIVE_MONITOR in monitor_indices
                ):
                    if locked_bar is None:
                        # Reacquire once from the whole monitor.
                        full_frame = grab_monitor(
                            sct,
                            ACTIVE_MONITOR,
                        )

                        if full_frame is None:
                            reset_tracker()
                            state = "WAIT_CAST"
                            post_catch_started_at = now
                            continue

                        reacquired = detect_minigame(full_frame)

                        if reacquired.get("valid"):
                            locked_bar = reacquired["bar"]
                            last_known_bar_by_monitor[
                                ACTIVE_MONITOR
                            ] = locked_bar
                            last_valid_minigame_at = now

                            print(
                                f"[TRACK] Locked bar {locked_bar} "
                                f"on M{ACTIVE_MONITOR}",
                                flush=True,
                            )
                        else:
                            if (
                                now - last_valid_minigame_at
                                >= MINIGAME_LOST_SECONDS
                            ):
                                reset_tracker()
                                state = "WAIT_CAST"
                                post_catch_started_at = now

                                print(
                                    "[STATE] FISHING -> WAIT_CAST "
                                    "(minigame ended/lost)",
                                    flush=True,
                                )

                            continue

                    fast_frame, local_bar, crop_origin = (
                        grab_locked_fishing_region(
                            sct,
                            ACTIVE_MONITOR,
                            locked_bar,
                        )
                    )

                    if fast_frame is None:
                        fast_track_failures += 1

                        if fast_track_failures >= 2:
                            release_movement()

                        if fast_track_failures >= FAST_TRACK_FAIL_LIMIT:
                            locked_bar = None

                        continue

                    local_ui = fast_track_minigame(
                        fast_frame,
                        local_bar,
                        previous_ui=previous_local_ui,
                    )

                    if not local_ui.get("valid"):
                        fast_track_failures += 1

                        # One missed frame is tolerated; repeated misses release.
                        if fast_track_failures >= 2:
                            release_movement()

                        if fast_track_failures >= 3:
                            previous_local_ui = None

                        if fast_track_failures >= FAST_TRACK_FAIL_LIMIT:
                            print(
                                f"[TRACK] Lost fish/green for "
                                f"{fast_track_failures} frames -> reacquire",
                                flush=True,
                            )
                            locked_bar = None

                        if (
                            now - last_valid_minigame_at
                            >= MINIGAME_LOST_SECONDS
                        ):
                            reset_tracker()
                            state = "WAIT_CAST"
                            post_catch_started_at = now

                            print(
                                "[STATE] FISHING -> WAIT_CAST "
                                "(round UI disappeared)",
                                flush=True,
                            )

                        continue

                    fast_track_failures = 0
                    last_valid_minigame_at = now

                    # Keep LOCAL previous geometry for the next high-speed frame.
                    previous_local_ui = dict(local_ui)

                    # Convert fast-crop boxes to monitor coordinates.
                    ui = dict(local_ui)

                    for key in ("bar", "fish", "zone"):
                        ui[key] = remap_box(
                            ui.get(key),
                            crop_origin,
                        )

                    # IMPORTANT: locked_bar is NOT rewritten every frame.
                    # It belongs to THIS round only and is replaced only when
                    # a fresh round is acquired after E.

                    fish_x = float(
                        ui["fish"][0]
                        + ui["fish"][2] / 2.0
                    )

                    green_x = float(
                        ui["zone"][0]
                        + ui["zone"][2] / 2.0
                    )

                    current_time = time.perf_counter()

                    # -----------------------------------------------------
                    # v10 DIRECT CURRENT-POSITION CONTROL
                    # -----------------------------------------------------
                    #
                    # v9's velocity prediction could briefly keep pointing in
                    # the OLD direction when the fish randomly reversed. That
                    # is exactly the behavior we do NOT want here.
                    #
                    # v10 therefore uses the current measured fish center and
                    # current measured green center only. Every new frame can
                    # reverse A <-> D immediately.
                    #
                    previous_fish_x = fish_x
                    previous_fish_time = current_time
                    fish_velocity = 0.0

                    predicted_fish_x = fish_x
                    lead = 0.0

                    error = (
                        fish_x
                        - green_x
                    )

                    zone_width = float(ui["zone"][2])

                    enter_band = max(
                        CONTROL_ENTER_MIN_PX,
                        zone_width * 0.035,
                    )

                    release_band = max(
                        CONTROL_RELEASE_MIN_PX,
                        enter_band * 0.38,
                    )

                    # -----------------------------------------------------
                    # v10 IMMEDIATE DIRECTION CONTROLLER
                    # -----------------------------------------------------
                    #
                    # fish left of green  -> A immediately
                    # fish right of green -> D immediately
                    # centers overlap     -> release
                    #
                    # There is no velocity state allowed to override the sign
                    # of the CURRENT measured error.
                    #
                    if error < -enter_band:
                        desired = "A"
                        drive("A")

                    elif error > enter_band:
                        desired = "D"
                        drive("D")

                    else:
                        desired = "NONE"
                        drive(None)

                    raw_inside = bool(
                        fish_x
                        >= ui["zone"][0]
                        + ui["zone"][2] * 0.04
                        and fish_x
                        <= ui["zone"][0]
                        + ui["zone"][2] * 0.96
                    )

                    # -----------------------------------------------------
                    # Progress is cheap; sample often.
                    # Ready template is heavier; sample less often.
                    # -----------------------------------------------------
                    if (
                        now - last_progress_check
                        >= PROGRESS_CHECK_INTERVAL
                    ):
                        (
                            cached_progress,
                            local_progress_box,
                        ) = detect_progress(
                            fast_frame,
                            local_bar,
                        )

                        cached_progress_box = remap_box(
                            local_progress_box,
                            crop_origin,
                        )

                        last_progress_check = now

                    if (
                        now - last_ready_check
                        >= READY_CHECK_INTERVAL
                    ):
                        (
                            cached_ready_score,
                            local_ready_box,
                            _,
                            cached_ready_ref,
                        ) = detect_ready(
                            fast_frame,
                            local_bar,
                        )

                        cached_ready_box = remap_box(
                            local_ready_box,
                            crop_origin,
                        )

                        last_ready_check = now

                    # Exact fish reference is already matched every
                    # active frame in v10. Save proof crops through the normal
                    # max-100/cooldown mechanism.
                    if (
                        ui.get("fish") is not None
                        and ui.get("fish_score", 0.0)
                        >= FISH_ACTIVE_MIN_SCORE
                    ):
                        local_fish_for_save = (
                            ui["fish"][0] - crop_origin[0],
                            ui["fish"][1] - crop_origin[1],
                            ui["fish"][2],
                            ui["fish"][3],
                        )

                        save_reference_hit(
                            fast_frame,
                            ACTIVE_MONITOR,
                            (
                                "fish-left.png"
                                if ui.get("fish_direction") == "left"
                                else "fish-right.png"
                            ),
                            ui.get("fish_score", 0.0),
                            local_fish_for_save,
                            FISH_ACTIVE_MIN_SCORE,
                        )

                    if ui.get("zone") is not None:
                        local_green_for_save = (
                            ui["zone"][0] - crop_origin[0],
                            ui["zone"][1] - crop_origin[1],
                            ui["zone"][2],
                            ui["zone"][3],
                        )

                        save_reference_hit(
                            fast_frame,
                            ACTIVE_MONITOR,
                            "green-target.png",
                            1.0,
                            local_green_for_save,
                            0.99,
                        )

                    if (
                        cached_ready_box is not None
                        and cached_ready_score
                        >= READY_THRESHOLD
                    ):
                        # Convert full-monitor ready box back to local crop for
                        # the exact hit image, avoiding a costly full-screen grab.
                        local_ready_for_save = (
                            cached_ready_box[0] - crop_origin[0],
                            cached_ready_box[1] - crop_origin[1],
                            cached_ready_box[2],
                            cached_ready_box[3],
                        )

                        save_reference_hit(
                            fast_frame,
                            ACTIVE_MONITOR,
                            cached_ready_ref
                            or "ready_space.png",
                            cached_ready_score,
                            local_ready_for_save,
                            READY_THRESHOLD,
                        )

                    # -----------------------------------------------------
                    # Console tracker telemetry
                    # -----------------------------------------------------
                    tracker_cycles += 1

                    if (
                        now - last_tracker_fps_log
                        >= 1.0
                    ):
                        elapsed_tracker = (
                            time.perf_counter()
                            - tracker_fps_started
                        )

                        measured_fps = (
                            tracker_cycles
                            / max(
                                elapsed_tracker,
                                0.001,
                            )
                        )

                        print(
                            f"[TRACK] round={round_number} "
                            f"M{ACTIVE_MONITOR} "
                            f"{measured_fps:.1f} scans/s "
                            f"fish={fish_x:.1f} "
                            f"green={green_x:.1f} "
                            f"lead={lead:+.1f} "
                            f"err={error:+.1f} "
                            f"key={desired} "
                            f"greenMode={ui.get('green_mode', '?')}",
                            flush=True,
                        )

                        tracker_cycles = 0
                        tracker_fps_started = (
                            time.perf_counter()
                        )
                        last_tracker_fps_log = now

                    if (
                        now - last_fish_log_at
                        >= FISH_LOG_INTERVAL
                    ):
                        print(
                            f"[FISH] round={round_number} "
                            f"fishX={fish_x:.1f} "
                            f"greenX={green_x:.1f} "
                            f"error={error:+.1f}px "
                            f"key={desired} "
                            f"greenMode={ui.get('green_mode', '?')} "
                            f"inside={'YES' if raw_inside else 'NO'} "
                            f"progress={cached_progress*100:.1f}% "
                            f"ready={cached_ready_score:.3f}/{READY_THRESHOLD:.2f}",
                            flush=True,
                        )

                        last_fish_log_at = now

                    # -----------------------------------------------------
                    # Catch - v11 delayed SPACE while tracking continues
                    # -----------------------------------------------------
                    #
                    # IMPORTANT:
                    # We do NOT release A/D when the ready prompt first appears.
                    # The green target keeps following the fish throughout the
                    # complete safety delay.
                    #
                    ready_visible = (
                        cached_ready_score
                        >= READY_THRESHOLD
                    )

                    if ready_visible:
                        if ready_visible_since is None:
                            ready_visible_since = time.perf_counter()

                            print(
                                f"[READY] Round {round_number}: "
                                f"prompt appeared -> starting "
                                f"{SPACE_READY_DELAY_SECONDS*1000:.0f}ms "
                                f"SPACE safety delay; A/D TRACKING CONTINUES",
                                flush=True,
                            )
                    else:
                        if ready_visible_since is not None:
                            print(
                                f"[READY] Round {round_number}: "
                                "prompt disappeared -> delay reset",
                                flush=True,
                            )

                        ready_visible_since = None

                    ready_elapsed = (
                        time.perf_counter() - ready_visible_since
                        if ready_visible_since is not None
                        else 0.0
                    )

                    if (
                        ready_visible
                        and ready_elapsed
                        >= SPACE_READY_DELAY_SECONDS
                        and raw_inside
                    ):
                        # Only NOW stop movement and press SPACE. Until this
                        # exact point the normal A/D controller above has kept
                        # the green target following the fish.
                        release_movement()

                        print(
                            f"[CATCH] Round {round_number}: "
                            f"ready visible for "
                            f"{ready_elapsed*1000:.0f}ms "
                            f"+ fish in green -> SPACE",
                            flush=True,
                        )

                        tap("SPACE")

                        reset_tracker()
                        state = "WAIT_CAST"
                        post_catch_started_at = now

                        # Permit the next cast immediately when cast.png appears.
                        last_cast_at = (
                            now
                            - CAST_DEBOUNCE_SECONDS
                            - 0.1
                        )

                        print(
                            "[STATE] -> WAIT_CAST "
                            "(next cast.png will trigger E immediately)",
                            flush=True,
                        )

                    elif (
                        ready_visible
                        and ready_visible_since is not None
                        and now - last_fish_log_at < 0.03
                    ):
                        # Compact telemetry so the console proves it is waiting
                        # deliberately rather than being stuck.
                        print(
                            f"[READY] delay="
                            f"{ready_elapsed*1000:.0f}/"
                            f"{SPACE_READY_DELAY_SECONDS*1000:.0f}ms "
                            f"inside={'YES' if raw_inside else 'NO'} "
                            f"trackingKey={desired}",
                            flush=True,
                        )

                    # Low-rate full monitor debug only.
                    if (
                        now - last_debug_at
                        >= DEBUG_SAVE_INTERVAL
                    ):
                        full_debug = grab_monitor(
                            sct,
                            ACTIVE_MONITOR,
                        )

                        if full_debug is not None:
                            debug_ui = dict(ui)
                            debug_ui["valid"] = True
                            debug_ui["progress"] = cached_progress
                            debug_ui["progress_box"] = cached_progress_box
                            debug_ui["ready_score"] = cached_ready_score
                            debug_ui["ready_box"] = cached_ready_box
                            debug_ui["predicted_fish_x"] = predicted_fish_x

                            save_debug(
                                full_debug,
                                state,
                                debug_ui,
                                last_scan.get(
                                    ACTIVE_MONITOR,
                                    {
                                        "waiting_score": 0.0,
                                        "waiting_box": None,
                                        "bite_score": 0.0,
                                        "bite_box": None,
                                        "cast_score": 0.0,
                                        "cast_box": None,
                                    },
                                ),
                                ACTIVE_MONITOR,
                            )

                        last_debug_at = now

                # =========================================================
                # SEARCH / ARMED / WAIT_CAST
                # =========================================================
                else:
                    release_movement()

                    # ARMED_FAST already did its own targeted acquisition above
                    # and is the one window where the bar is about to appear.
                    # Stock v11 fell through to here anyway and spent ~800 ms
                    # per scan on detect_minigame + waiting + bite + cast over
                    # the whole screen, which is what made the wait blind.
                    # E has been pressed, so there is nothing here it needs.
                    if (
                        state == "ARMED_FAST"
                        and ARMED_SKIP_FULL_SCAN
                        and now - armed_started_at
                        < ARMED_SKIP_FULL_SCAN_SECONDS
                    ):
                        pass

                    elif (
                        now - last_scan_at
                        >= SCAN_INTERVAL
                    ):
                        last_scan_at = now

                        ordered = list(monitor_indices)

                        if ACTIVE_MONITOR in ordered:
                            ordered.remove(ACTIVE_MONITOR)
                            ordered.insert(
                                0,
                                ACTIVE_MONITOR,
                            )

                        for mi in ordered:
                            frame = grab_monitor(
                                sct,
                                mi,
                            )

                            if frame is None:
                                continue

                            # ------------------------------------------------
                            # After a completed catch, ONLY cast.png matters.
                            # This prevents stale previous-round minigame pixels
                            # from being treated as a new round.
                            # ------------------------------------------------
                            if state == "WAIT_CAST":
                                if (
                                    now - post_catch_started_at
                                    < POST_CATCH_MIN_DELAY
                                ):
                                    continue

                                (
                                    cast_score,
                                    cast_box,
                                    cast_scale,
                                ) = detect_cast(frame)

                                if (
                                    cast_score
                                    >= CAST_THRESHOLD
                                    and (
                                        now - last_cast_at
                                        >= CAST_DEBOUNCE_SECONDS
                                    )
                                ):
                                    save_reference_hit(
                                        frame,
                                        mi,
                                        "cast.png",
                                        cast_score,
                                        cast_box,
                                        CAST_THRESHOLD,
                                    )

                                    start_new_cast(
                                        sct,
                                        mi,
                                        cast_box,
                                        cast_score,
                                        cast_scale,
                                    )

                                    break

                                if (
                                    now - last_status_at
                                    >= STATUS_LOG_INTERVAL
                                ):
                                    print(
                                        f"[WAIT_CAST] M{mi} "
                                        f"cast={cast_score:.3f}/"
                                        f"{CAST_THRESHOLD:.2f}",
                                        flush=True,
                                    )

                                continue

                            # ------------------------------------------------
                            # IDLE/ARMED: acquire fresh minigame.
                            # ------------------------------------------------
                            ui = detect_minigame(frame)

                            if (
                                ui.get("valid")
                                and state == "IDLE"
                            ):
                                ACTIVE_MONITOR = mi

                                target_window_from_detection(
                                    sct,
                                    mi,
                                    ui["bar"],
                                )

                                reset_tracker()

                                locked_bar = ui["bar"]
                                last_known_bar_by_monitor[mi] = locked_bar
                                last_valid_minigame_at = now

                                print(
                                    f"[STATE] Round {round_number}: "
                                    f"-> FISHING on M{mi}",
                                    flush=True,
                                )

                                print(
                                    f"[DETECT] initial fish refs: "
                                    f"L={ui['fish_left_score']:.3f} "
                                    f"R={ui['fish_right_score']:.3f} "
                                    f"chosen={ui['fish_direction']}",
                                    flush=True,
                                )

                                state = "FISHING"

                                break

                            # ------------------------------------------------
                            # Text references / cast
                            # ------------------------------------------------
                            (
                                waiting_score,
                                waiting_box,
                                waiting_scale,
                            ) = detect_waiting(frame)

                            (
                                bite_score,
                                bite_box,
                                bite_scale,
                            ) = detect_bite(frame)

                            (
                                cast_score,
                                cast_box,
                                cast_scale,
                            ) = detect_cast(frame)

                            scan = {
                                "waiting_score": waiting_score,
                                "waiting_box": waiting_box,
                                "waiting_scale": waiting_scale,
                                "bite_score": bite_score,
                                "bite_box": bite_box,
                                "bite_scale": bite_scale,
                                "cast_score": cast_score,
                                "cast_box": cast_box,
                                "cast_scale": cast_scale,
                            }

                            last_scan[mi] = scan

                            save_reference_hit(
                                frame,
                                mi,
                                "waiting.png",
                                waiting_score,
                                waiting_box,
                                WAITING_THRESHOLD,
                            )

                            save_reference_hit(
                                frame,
                                mi,
                                "bite.png",
                                bite_score,
                                bite_box,
                                BITE_THRESHOLD,
                            )

                            save_reference_hit(
                                frame,
                                mi,
                                "cast.png",
                                cast_score,
                                cast_box,
                                CAST_THRESHOLD,
                            )

                            if (
                                cast_score
                                >= CAST_THRESHOLD
                                and (
                                    now - last_cast_at
                                    >= CAST_DEBOUNCE_SECONDS
                                )
                            ):
                                start_new_cast(
                                    sct,
                                    mi,
                                    cast_box,
                                    cast_score,
                                    cast_scale,
                                )
                                break

                            if (
                                now - last_status_at
                                >= STATUS_LOG_INTERVAL
                            ):
                                print(
                                    f"[SCAN] M{mi} state={state} "
                                    f"wait={waiting_score:.3f} "
                                    f"bite={bite_score:.3f} "
                                    f"cast={cast_score:.3f}/"
                                    f"{CAST_THRESHOLD:.2f} "
                                    f"bar={'YES' if ui.get('bar') else 'NO'} "
                                    f"zone={'YES' if ui.get('zone') else 'NO'}",
                                    flush=True,
                                )

                        if (
                            now - last_status_at
                            >= STATUS_LOG_INTERVAL
                        ):
                            last_status_at = now

                # =========================================================
                # Loop limiting
                # =========================================================
                elapsed = (
                    time.perf_counter()
                    - loop_started
                )

                # Active FISHING is deliberately uncapped.
                if not (
                    (
                        state == "FISHING"
                        and TRACK_UNCAPPED
                    )
                    or (
                        state == "ARMED_FAST"
                        and FAST_ARMED_UNCAPPED
                    )
                ):
                    wait = (
                        1.0 / TARGET_FPS
                        - elapsed
                    )

                    if wait > 0:
                        time.sleep(wait)

        except KeyboardInterrupt:
            print("\n[STATE] Ctrl+C -> EXIT", flush=True)

        finally:
            release_movement()
            print("[DONE] A/D released.", flush=True)


if __name__ == "__main__":
    main()
